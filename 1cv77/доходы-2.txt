Функция ПолучитьНомерСчетаИзКомментария(Стр)
    НомерСчета = "";
    ОстатокСтроки = "";

    // Вариант 1: поиск по символу "№" (напр. "Счет на оплату № 00536")
    ПозНомера = Найти(Стр, "№");
    Если ПозНомера > 0 Тогда
        ОстатокСтроки = СокрЛП(Сред(Стр, ПозНомера + 1));
    КонецЕсли;

    // Вариант 2: поиск по "сч " (напр. "Оплата по сч 536")
    // Проверяем только если по "№" ничего не нашли
    Если ОстатокСтроки = "" Тогда
        // Ищем "сч " без учета регистра — перебираем варианты написания
        
        //Маски[1] = "сч ";
        //Маски[2] = "Сч ";
        //Маски[3] = "СЧ ";
        //Маски[4] = "сЧ "; 
		Маска = "сч ";
		
		позсч = Найти(ВРЕГ(Стр), ВРЕГ(Маска));
        
		Если позсч > 0 тогда
			ОстатокСтроки = СокрЛП(Сред(Стр, позсч + СтрДлина(Маска)));
		конецЕсли;
    КонецЕсли;

    Если ОстатокСтроки = "" Тогда
        Возврат "";
    КонецЕсли;

    // Извлекаем номер — всё до первого нецифрового символа
    // (защита от "от", ",", пробела и т.д.)
    Для й = 1 По СтрДлина(ОстатокСтроки) Цикл
        Символ = Сред(ОстатокСтроки, й, 1);
        Если Найти("0123456789", Символ) > 0 Тогда
            НомерСчета = НомерСчета + Символ;
        Иначе
            Если НомерСчета <> "" Тогда
                // Уже начали читать цифры — первый не-цифровой означает конец номера
                Прервать;
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;

    Возврат НомерСчета;
КонецФункции

Функция ПолучитьНомерИДатуРСдляУПДИзСтрокиПервичногоДокумента(Стр, номеррасч, датарасчт)
    НомерДок = "";
    ДатаДок  = "";

    //--- Номер: цифры после "р/с " ---
    Поз = Найти(Стр, "р/с ");
    Если Поз = 0 Тогда Поз = Найти(Стр, "Р/с "); КонецЕсли;
    Если Поз = 0 Тогда Поз = Найти(Стр, "Р/С "); КонецЕсли;
    Если Поз > 0 Тогда
        Остаток = СокрЛП(Сред(Стр, Поз + 4));
        Для й = 1 По СтрДлина(Остаток) Цикл
            Симвл = Сред(Остаток, й, 1);
            Если Найти("0123456789", Симвл) > 0 Тогда
                НомерДок = НомерДок + Симвл;
            Иначе
                Если НомерДок <> "" Тогда Прервать; КонецЕсли;
            КонецЕсли;
        КонецЦикла;
    КонецЕсли;

    //--- Дата: содержимое скобок ---
    ПозОтк = Найти(Стр, "(");
    ПозЗкр = Найти(Стр, ")");
    Если (ПозОтк > 0) И (ПозЗкр > ПозОтк) Тогда
        ДатаДок = СокрЛП(Сред(Стр, ПозОтк + 1, ПозЗкр - ПозОтк - 1));
    КонецЕсли;

    //--- Результат ---
    Если НомерДок = "" Тогда Возврат ""; КонецЕсли;
    Если ДатаДок  = "" Тогда Возврат НомерДок; КонецЕсли;            
	номеррасч = НомерДок; датарасчт = ДатаДок;
    Возврат НомерДок + " от " + ДатаДок;

Функция РНП_ПолеТаб(Стр, НомерПоля)
    Остаток = Стр;
    Ном = 1;
    Таб = Симв(9);
    Пока 1 = 1 Цикл
        П = Найти(Остаток, Таб);
        Если П = 0 Тогда
            Если Ном = НомерПоля Тогда
                Возврат СокрЛП(Остаток);
            Иначе
                Возврат "";
            КонецЕсли;
        КонецЕсли;
        Если Ном = НомерПоля Тогда
            Возврат СокрЛП(Лев(Остаток, П - 1));
        КонецЕсли;
        Остаток = Сред(Остаток, П + 1, СтрДлина(Остаток));
        Ном = Ном + 1;
    КонецЦикла;

// В 1С 7.7 нет СтрЧислоСтрок / СтрПолучитьСтроку (это 8.x).
Функция РНП_РазобратьОтвет(Ответ, Смысл, ТаблицаДокументов)
    ПС = Симв(10);
    Норм = СтрЗаменить(Ответ, Симв(13) + ПС, ПС);
    Норм = СтрЗаменить(Норм, Симв(13), ПС);

    Первая = 1;
    Остаток = Норм;
    ЕстьЕще = 1;
    Пока ЕстьЕще = 1 Цикл
        П = Найти(Остаток, ПС);
        Если П = 0 Тогда
            ТекСтрока = СокрЛП(Остаток);
            ЕстьЕще = 0;
        Иначе
            ТекСтрока = СокрЛП(Лев(Остаток, П - 1));
            Остаток = Сред(Остаток, П + 1, СтрДлина(Остаток));
        КонецЕсли;

        Если ПустаяСтрока(ТекСтрока) = 0 Тогда
            Если Первая = 1 Тогда
                КодСтроки = РНП_ПолеТаб(ТекСтрока, 1);
                Если КодСтроки = "ERROR" Тогда
                    Возврат 0;
                КонецЕсли;
                Если КодСтроки <> "OK" Тогда
                    Возврат 0;
                КонецЕсли;
                Смысл = РНП_ПолеТаб(ТекСтрока, 2);
                Если ПустаяСтрока(Смысл) = 1 Тогда
                    Смысл = "НЕОПРЕДЕЛЕНО";
                КонецЕсли;
                Первая = 0;
            Иначе
                Если РНП_ПолеТаб(ТекСтрока, 1) = "DOC" Тогда
                    ТипДок = РНП_ПолеТаб(ТекСтрока, 2);
                    НомерДок = РНП_ПолеТаб(ТекСтрока, 3);
                    ДатаДок = РНП_ПолеТаб(ТекСтрока, 4);
                    Если (ПустаяСтрока(ТипДок) = 0) И ((ПустаяСтрока(НомерДок) = 0) ИЛИ (ПустаяСтрока(ДатаДок) = 0)) Тогда
                        ТаблицаДокументов.НоваяСтрока();
                        ТаблицаДокументов.ТипДок = ТипДок;
                        ТаблицаДокументов.НомерДок = НомерДок;
                        ТаблицаДокументов.ДатаДок = ДатаДок;
                    КонецЕсли;
                КонецЕсли;
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;

    Если Первая = 1 Тогда
        Возврат 0;
    КонецЕсли;
    Возврат 1;
