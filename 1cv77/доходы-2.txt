Функция ПолучитьНомерСчетаИзКомментария(Стр)
    НомерСчета = "";
    ОстатокСтроки = "";

    // Вариант 1: поиск по символу "№" (напр. "Счет на оплату № 00536")
    ПозНомера = Найти(Стр, "№");
    Если ПозНомера > 0 Тогда
        ОстатокСтроки = СокрЛП(Сред(Стр, ПозНомера + 1));
    КонецЕсли;

    // Вариант 2: поиск по "сч " (напр. "Оплата по сч 536")
    // Проверяем только если по "№" ничего не нашли
    Если ОстатокСтроки = "" Тогда
        // Ищем "сч " без учета регистра — перебираем варианты написания
        
        //Маски[1] = "сч ";
        //Маски[2] = "Сч ";
        //Маски[3] = "СЧ ";
        //Маски[4] = "сЧ "; 
		Маска = "сч ";
		
		позсч = Найти(ВРЕГ(Стр), ВРЕГ(Маска));
        
		Если позсч > 0 тогда
			ОстатокСтроки = СокрЛП(Сред(Стр, позсч + СтрДлина(Маска)));
		конецЕсли;
    КонецЕсли;

    Если ОстатокСтроки = "" Тогда
        Возврат "";
    КонецЕсли;

    // Извлекаем номер — всё до первого нецифрового символа
    // (защита от "от", ",", пробела и т.д.)
    Для й = 1 По СтрДлина(ОстатокСтроки) Цикл
        Символ = Сред(ОстатокСтроки, й, 1);
        Если Найти("0123456789", Символ) > 0 Тогда
            НомерСчета = НомерСчета + Символ;
        Иначе
            Если НомерСчета <> "" Тогда
                // Уже начали читать цифры — первый не-цифровой означает конец номера
                Прервать;
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;

    Возврат НомерСчета;
КонецФункции

    ДатаДок  = "";

    Поз = Найти(Стр, "р/с ");
    Если Поз = 0 Тогда Поз = Найти(Стр, "Р/с "); КонецЕсли;
    Если Поз = 0 Тогда Поз = Найти(Стр, "Р/С "); КонецЕсли;

        Остаток = СокрЛП(Сред(Стр, Поз + 4));
        Для й = 1 По СтрДлина(Остаток) Цикл
            Симвл = Сред(Остаток, й, 1);

            Иначе

            КонецЕсли;
        КонецЦикла;

    //--- Дата: содержимое скобок ---
