// Возвращает остаток суммы, которую не удалось погасить,
// потому что открытой задолженности уже нет.
//
// ТЗДолги:
//   Документ
//   ДатаДок
//   СуммаДолга
//   Остаток
//
Функция ПогаситьДолгиFIFO(ТЗДолги, СуммаПогашения)

    Если СуммаПогашения <= 0 Тогда
        Возврат 0;
    КонецЕсли;

    Для НомСтр = 1 По ТЗДолги.КоличествоСтрок() Цикл

        Остаток = ТЗДолги.ПолучитьЗначение(НомСтр, "Остаток");

        Если Остаток > 0 Тогда

            Если СуммаПогашения >= Остаток Тогда

                // Этот документ закрываем полностью
                ТЗДолги.УстановитьЗначение(
                    НомСтр,
                    "Остаток",
                    0
                );

                СуммаПогашения = СуммаПогашения - Остаток;

            Иначе

                // Документ закрываем частично
                ТЗДолги.УстановитьЗначение(
                    НомСтр,
                    "Остаток",
                    Остаток - СуммаПогашения
                );

                СуммаПогашения = 0;

            КонецЕсли;

            Если СуммаПогашения <= 0 Тогда
                Прервать;
            КонецЕсли;

        КонецЕсли;

    КонецЦикла;

    Возврат СуммаПогашения;

КонецФункции


//*******************************************************
//
// Определяет, какие документы задолженности закрывает
// текущая оплата.
//
// СуммаОплатыВперед — оплата, которую сначала зачитываем
// по тем же долгам FIFO. Документы по ней не возвращаются.
// Если после зачёта остаётся незакрытая задолженность,
// её закрывает СуммаОплаты; в результат попадают только
// документы, которые закрыла именно СуммаОплаты.
//
// Возвращает ТаблицуЗначений:
//
// Документ
// ДатаДок
// ОстатокДоОплаты
// СуммаПогашения
// ОстатокПослеОплаты
//
Функция ПолучитьДокументыОплатыЗадолжностиКлиента(
        Клиент,
        ДокументОплаты,
        СуммаОплатыВперед,
        СуммаОплаты)

    // -------------------------------------------------
    // Результат
    // -------------------------------------------------

    ТЗРезультат = СоздатьОбъект("ТаблицаЗначений");

    ТЗРезультат.НоваяКолонка("Документ");
    ТЗРезультат.НоваяКолонка("ДатаДок", "Дата");
    ТЗРезультат.НоваяКолонка("ОстатокДоОплаты", "Число", 15, 2);
    ТЗРезультат.НоваяКолонка("СуммаПогашения", "Число", 15, 2);
    ТЗРезультат.НоваяКолонка("ОстатокПослеОплаты", "Число", 15, 2);


    Если СуммаОплаты <= 0 Тогда
        Возврат ТЗРезультат;
    КонецЕсли;


    // -------------------------------------------------
    // Здесь будем восстанавливать текущие долги
    // по документам.
    // -------------------------------------------------

    ТЗДолги = СоздатьОбъект("ТаблицаЗначений");

    ТЗДолги.НоваяКолонка("Документ");
    ТЗДолги.НоваяКолонка("ДатаДок", "Дата");
    ТЗДолги.НоваяКолонка("СуммаДолга", "Число", 15, 2);
    ТЗДолги.НоваяКолонка("Остаток", "Число", 15, 2);


    // -------------------------------------------------
    // Счет задолженности покупателей.
    //
    // В типовой Бухгалтерии 7.7 это обычно 62.1.
    // Если у вас используется другой счет -
    // поменять здесь.
    // -------------------------------------------------

    СчетДолга = СчетПоКоду("62.1");


    // -------------------------------------------------
    // Берем проводки СТРОГО ДО текущего документа оплаты.
    //
    // Это важно: документы той же даты, проведенные
    // после ДокументОплаты, сюда попадать не должны.
    // -------------------------------------------------

    ПредыдущаяПозиция =
        СформироватьПозициюДокумента(
            ДокументОплаты.ПолучитьПозицию(),
            -1
        );


    Опер = СоздатьОбъект("Операция");

    // Только наш клиент
    Опер.ИспользоватьСубконто(
        ВидыСубконто.Контрагенты,
        Клиент
    );


    // -------------------------------------------------
    // Выбираем все движения по 62.1 до оплаты.
    //
    // Флаг 3 = и дебет, и кредит счета.
    // -------------------------------------------------

    Опер.ВыбратьОперацииСПроводками(
        ,
        ПредыдущаяПозиция,
        СчетДолга,
        ,
        3
    );


    // Если когда-то возник кредитовый остаток по 62.1,
    // который нечем было закрыть, запоминаем его.
    СвободныйКредит = 0;


    Пока Опер.ПолучитьПроводку() = 1 Цикл

        ТекДокумент = Опер.ПолучитьДокумент();


        // =============================================
        // ДЕБЕТ 62.1
        //
        // Возникла задолженность клиента.
        // Например:
        //
        //   Дт 62.1 Кт 90...
        //
        // =============================================

        Если Опер.Дебет.Счет = СчетДолга Тогда

            СуммаДолга = Опер.Сумма;

            Если СуммаДолга > 0 Тогда

                НомСтрокиДок = 0;

                Если ТЗДолги.НайтиЗначение(
                        ТекДокумент,
                        НомСтрокиДок,
                        "Документ"
                   ) = 1 Тогда

                    // У одного документа может быть
                    // несколько проводок Дт 62.1

                    СтарыйДолг =
                        ТЗДолги.ПолучитьЗначение(
                            НомСтрокиДок,
                            "СуммаДолга"
                        );

                    СтарыйОстаток =
                        ТЗДолги.ПолучитьЗначение(
                            НомСтрокиДок,
                            "Остаток"
                        );

                    ТЗДолги.УстановитьЗначение(
                        НомСтрокиДок,
                        "СуммаДолга",
                        СтарыйДолг + СуммаДолга
                    );

                    ТЗДолги.УстановитьЗначение(
                        НомСтрокиДок,
                        "Остаток",
                        СтарыйОстаток + СуммаДолга
                    );

                Иначе

                    ТЗДолги.НоваяСтрока();

                    ТЗДолги.Документ = ТекДокумент;
                    ТЗДолги.ДатаДок = Опер.ДатаОперации;
                    ТЗДолги.СуммаДолга = СуммаДолга;
                    ТЗДолги.Остаток = СуммаДолга;

                КонецЕсли;


                // Если до этого был кредитовый остаток,
                // погашаем им появившийся долг.
                Если СвободныйКредит > 0 Тогда

                    СвободныйКредит =
                        ПогаситьДолгиFIFO(
                            ТЗДолги,
                            СвободныйКредит
                        );

                КонецЕсли;

            КонецЕсли;

        КонецЕсли;


        // =============================================
        // КРЕДИТ 62.1
        //
        // Задолженность уменьшается.
        // Например:
        //
        //   Дт 51 Кт 62.1
        //
        // Это предыдущая оплата клиента.
        //
        // =============================================

        Если Опер.Кредит.Счет = СчетДолга Тогда

            СуммаЗакрытия = Опер.Сумма;

            Если СуммаЗакрытия > 0 Тогда

                СвободныйКредит =
                    ПогаситьДолгиFIFO(
                        ТЗДолги,
                        СвободныйКредит + СуммаЗакрытия
                    );

            КонецЕсли;

        КонецЕсли;


    КонецЦикла;


    // -------------------------------------------------
    // На этом этапе ТЗДолги содержит состояние долгов
    // непосредственно ПЕРЕД ДокументОплаты.
    //
    // Сначала зачитываем оплату вперёд.
    // Алгоритм тот же (FIFO), но документы по ней
    // в результат не попадают.
    // -------------------------------------------------

    ПогаситьДолгиFIFO(ТЗДолги, СуммаОплатыВперед);


    // -------------------------------------------------
    // Теперь распределяем именно СуммаОплаты
    // по тому, что ещё осталось неоплаченным.
    // -------------------------------------------------

    ОсталосьРаспределить = СуммаОплаты;


    Для НомСтр = 1 По ТЗДолги.КоличествоСтрок() Цикл

        Если ОсталосьРаспределить <= 0 Тогда
            Прервать;
        КонецЕсли;


        ОстатокДолга =
            ТЗДолги.ПолучитьЗначение(
                НомСтр,
                "Остаток"
            );


        // Полностью закрытые предыдущими оплатами
        // документы нас уже не интересуют.
        Если ОстатокДолга > 0 Тогда

            Если ОсталосьРаспределить >= ОстатокДолга Тогда

                СуммаНаДокумент = ОстатокДолга;

            Иначе

                СуммаНаДокумент = ОсталосьРаспределить;

            КонецЕсли;


            // -----------------------------------------
            // Добавляем документ в результат
            // -----------------------------------------

            ТЗРезультат.НоваяСтрока();

            ТЗРезультат.Документ =
                ТЗДолги.ПолучитьЗначение(
                    НомСтр,
                    "Документ"
                );

            ТЗРезультат.ДатаДок =
                ТЗДолги.ПолучитьЗначение(
                    НомСтр,
                    "ДатаДок"
                );

            ТЗРезультат.ОстатокДоОплаты =
                ОстатокДолга;

            ТЗРезультат.СуммаПогашения =
                СуммаНаДокумент;

            ТЗРезультат.ОстатокПослеОплаты =
                ОстатокДолга - СуммаНаДокумент;


            ОсталосьРаспределить =
                ОсталосьРаспределить - СуммаНаДокумент;

        КонецЕсли;

    КонецЦикла;


    Возврат ТЗРезультат;
