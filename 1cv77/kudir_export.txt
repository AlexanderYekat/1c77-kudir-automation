Функция КУДиР_Выполнить(датанач, датакон, уровеньЛога)
	глОшибка = 0;
	глЛогЗеркало = 1;
	глЛогПуть = "";
	глCSVБуфер = "";
	глУровеньЛога = 2;
	глПотокCSVОткрыт = 0;
	глПотокIdMapОткрыт = 0;
	глСтрокCSV = 0;
	глСтрокIdMap = 0;
	глИмяПотокаCSV = "";
	глЭтапВыгрузки = "init";
	КУДиР_УстановитьУровеньЛога(уровеньЛога);
	КУДиР_Состояние("КУДиР: инициализация");
	КУДиР_ИнициализироватьПути();
	Если глОшибка = 1 Тогда
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;
	КУДиР_НачатьЛог();
	глRegisteredIds = КУДиР_НовыйСловарь();
	глКлиентыИд = КУДиР_НовыйСловарь();
	глPaymentsIdx = КУДиР_НовыйСловарь();
	глPaymentIds = КУДиР_НовыйСловарь();
	глDocuments = КУДиР_НовыйСловарь();
	глLedgerIds = КУДиР_НовыйСловарь();
	глРеквКэш = КУДиР_НовыйСловарь();
	глВидыДокКэш = КУДиР_НовыйСловарь();
	глМесяцевИстории = 3;
	глМесяцевГоризонта = 3;
	глПослПозиция = "";
	глПослПроводка = "";
	глНомКорр = 0;

	историяНач = ДобавитьМесяц(КУДиР_НачалоГода(датанач), -глМесяцевИстории);
	квКонец = КУДиР_КонецКвартала(датакон);
	горизонт = КУДиР_КонецКвартала(ДобавитьМесяц(датакон, глМесяцевГоризонта));
	КУДиР_Этап("КУДиР: инициализация");
	КУДиР_Лог("уровень лога " + КУДиР_ИмяУровня(глУровеньЛога) + " (" + СокрЛП(глУровеньЛога) + ")");
	КУДиР_Лог("ДатаНач=" + КУДиР_ДатаISO(датанач) + " ДатаКон=" + КУДиР_ДатаISO(датакон) + " ИсторияНач=" + КУДиР_ДатаISO(историяНач) + " КварталКонец=" + КУДиР_ДатаISO(квКонец) + " MatchHorizonEnd=" + КУДиР_ДатаISO(горизонт));

	таблКлиенты = КУДиР_СоздатьТаблицуКлиентов();
	глЭтапВыгрузки = "проход A";
	КУДиР_Этап("КУДиР: проход A, целевые клиенты 50/51/55 -> 62");
	КУДиР_СобратьКлиентов(датанач, датакон, таблКлиенты);
	Если глОшибка = 1 Тогда
		КУДиР_ОсвободитьЭкспорт();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;
	Если таблКлиенты.КоличествоСтрок() = 0 Тогда
		КУДиР_ЛогУр(1, "КУДиР выгрузка: нет целевых оплат 50/51/55 -> 62 за ДатаНач...ДатаКон.");
		КУДиР_ОсвободитьЭкспорт();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;
	КУДиР_Лог("клиентов=" + СокрЛП(таблКлиенты.КоличествоСтрок()));

	run_id = КУДиР_НовыйRunId(датанач);
	Если (глОшибка = 1) Или (СокрЛП(run_id) = "") Тогда
		КУДиР_ОсвободитьЭкспорт();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;
	FSO = СоздатьОбъект("Scripting.FileSystemObject");
	глКаталогОбмена = глКаталогПроекта + "\exchange\" + run_id;
	Если FSO.FolderExists(глКаталогОбмена) = 0 Тогда
		FSO.CreateFolder(глКаталогОбмена);
	КонецЕсли;
	КУДиР_ПеренестиЛогВКаталогЗапуска();
	КУДиР_Лог("каталог запуска " + глКаталогОбмена);
	КУДиР_ЗаписатьВыгрузку(run_id, историяНач, датанач, датакон, квКонец, горизонт);
	КУДиР_ОткрытьIdMap();
	КУДиР_ЗаписатьСтатусЭкспорта("IN_PROGRESS");
	Если глОшибка = 1 Тогда
		КУДиР_ЗавершитьВыгрузкуОшибкой();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;

	таблОплаты = КУДиР_СоздатьТаблицуОплат();
	глЭтапВыгрузки = "проход B";
	КУДиР_Этап("КУДиР: проход B, история оплат выбранных клиентов");
	КУДиР_СобратьОплаты(историяНач, датанач, датакон, таблОплаты, таблКлиенты);
	Если глОшибка = 1 Тогда
		КУДиР_ЗавершитьВыгрузкуОшибкой();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;
	КУДиР_Лог("оплат=" + СокрЛП(таблОплаты.КоличествоСтрок()));
	КУДиР_ЗаписатьPayments(таблОплаты);
	таблОплаты = 0;
	глPaymentsIdx = 0;
	Если глОшибка = 1 Тогда
		КУДиР_ЗавершитьВыгрузкуОшибкой();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;

	глЭтапВыгрузки = "ledger";
	КУДиР_Этап("КУДиР: поток ledger62");
	КУДиР_СобратьLedger(историяНач, датакон, таблКлиенты);
	Если глОшибка = 1 Тогда
		КУДиР_ЗавершитьВыгрузкуОшибкой();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;

	глЭтапВыгрузки = "documents";
	КУДиР_Этап("КУДиР: поток documents + НДС");
	КУДиР_СобратьДокументы(историяНач, горизонт, таблКлиенты);
	Если глОшибка = 1 Тогда
		КУДиР_ЗавершитьВыгрузкуОшибкой();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;

	глЭтапВыгрузки = "saldo";
	КУДиР_Этап("КУДиР: поток opening_balances на ИсторияНач");
	КУДиР_СобратьСальдо(историяНач, таблКлиенты);
	Если глОшибка = 1 Тогда
		КУДиР_ЗавершитьВыгрузкуОшибкой();
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;

	КУДиР_ЗакрытьВсеПотоки();
	КУДиР_ЗаписатьСтатусЭкспорта("SUCCESS");
	КУДиР_Лог("export SUCCESS, id_map строк " + СокрЛП(глСтрокIdMap));
	КУДиР_ОсвободитьЭкспорт();
	таблКлиенты = 0;

	КУДиР_Этап("КУДиР: запуск Python matching");
	Код = КУДиР_ЗапуститьPython();
	КУДиР_ЛогФайл(3, "Python код возврата " + СокрЛП(Код));
	Если Код <> 0 Тогда
		КУДиР_ЛогУр(1, "КУДиР: Python вернул код " + СокрЛП(Код) + ". Смотрите " + глКаталогОбмена + "\\run_status.csv и python_run.log");
	КонецЕсли;

	КУДиР_Этап("КУДиР: загрузка результата");
	табРез = КУДиР_ЗагрузитьИВосстановить();
	КУДиР_Этап("КУДиР: готово");
	Возврат табРез;

Функция КУДиР_ИмяУровня(Знач Ур)
	Если Число(Ур) <= 0 Тогда
		Возврат "ERROR";
	КонецЕсли;
	Если Число(Ур) = 1 Тогда
		Возврат "WARN";
	КонецЕсли;
	Если Число(Ур) = 2 Тогда
		Возврат "INFO";

Функция КУДиР_ПолеCSV(Знач С)
	С = "" + С;
	Кавычка = Симв(34);
	НужныКавычки = 0;
	Если Найти(С, ";") > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Кавычка) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Симв(10)) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Симв(13)) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	// Начальные/конечные пробелы и табуляция: поле в кавычках, чтобы читатель не обрезал позицию (№1 §30).
	Если СтрДлина(С) > 0 Тогда
		КодПерв = КодСимв(Лев(С, 1));
		КодПосл = КодСимв(Прав(С, 1));
		Если (КодПерв = 32) Или (КодПерв = 9) Или (КодПосл = 32) Или (КодПосл = 9) Тогда
			НужныКавычки = 1;
		КонецЕсли;
	КонецЕсли;
	Если НужныКавычки = 1 Тогда
		С = СтрЗаменить(С, Кавычка, Кавычка + Кавычка);
		Возврат Кавычка + С + Кавычка;
	КонецЕсли;
	Возврат С;

Функция КУДиР_ЕстьВидДокумента(Знач ИмяВида)

	// вызов Получить() у числа - ошибка исполнения. Используется только отключённой выгрузкой v2;
	// КУДиР_ЕстьРеквизитДокумента опирается на неё. Блок КУДиР4_ метаданные не опрашивает.

					Найден = 1;
				КонецЕсли;

Функция КУДиР_ВидОплаты(Знач ВидДок)
	// ПКО этой конфигурации: идентификатор вида «ПКО».
	// Обе стороны сравнения нормализованы Врег. Не искать смешанный регистр внутри Врег(ВидДок).
	Норм = Врег(СокрЛП(ВидДок));
	Если (Норм = Врег("ПКО")) Или (Норм = Врег("ПриходныйОрдер")) Или (Норм = Врег("ПриходныйКассовыйОрдер")) Тогда
		Возврат "CASH";
	КонецЕсли;
	Возврат "BANK";

Процедура КУДиР_СобратьКлиентов(датанач, датакон, таблКлиенты)
	Опер = СоздатьОбъект("Операция");
	ДокОбъект = СоздатьОбъект("Документ");
	СписокПроводок = "50,62;51,62;55,62;";
	Опер.ВыбратьОперацииСПроводками(датанач, датакон, СписокПроводок);
	нПров = 0;
	Пока Опер.ПолучитьПроводку() = 1 Цикл
		нПров = нПров + 1;
		Если (нПров = 1) Или (Цел(нПров / 50) * 50 = нПров) Тогда
			КУДиР_Состояние("Проход A: проводок " + СокрЛП(нПров) + ", клиентов " + СокрЛП(таблКлиенты.КоличествоСтрок()));
		КонецЕсли;
		Если КУДиР_ЭтоСчет62(Опер.Кредит.Счет) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Если КУДиР_ЭтоДенежныйСчет(Опер.Дебет.Счет) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Док = "";
		Если ДокОбъект.НайтиДокумент(Опер.Документ) = 1 Тогда
			Док = ДокОбъект.ТекущийДокумент();
		КонецЕсли;
		Если ПустоеЗначение(Док) = 1 Тогда
			КУДиР_Сбой("проводка оплаты без документа операции, дата=" + КУДиР_ДатаISO(Опер.ДатаОперации) + ", сумма=" + СокрЛП(Опер.Сумма));
			Продолжить;
		КонецЕсли;
		Контр = Опер.Кредит.Контрагенты;
		cid = КУДиР_ИдКонтрагента(Контр);
		КУДиР_ДобавитьКлиента(таблКлиенты, cid, Контр);
	КонецЦикла;
	КУДиР_Этап("Проход A: готово, проводок " + СокрЛП(нПров) + ", клиентов " + СокрЛП(таблКлиенты.КоличествоСтрок()));
КонецПроцедуры

Процедура КУДиР_СобратьОплаты(историяНач, датанач, датакон, таблОплаты, таблКлиенты)
	Опер = СоздатьОбъект("Операция");
	ДокОбъект = СоздатьОбъект("Документ");
	СписокПроводок = "50,62;51,62;55,62;";
	Опер.ВыбратьОперацииСПроводками(историяНач, датакон, СписокПроводок);
	нПров = 0;
	Пока Опер.ПолучитьПроводку() = 1 Цикл
		нПров = нПров + 1;
		Если (нПров = 1) Или (Цел(нПров / 50) * 50 = нПров) Тогда
			КУДиР_Состояние("Проход B: проводок " + СокрЛП(нПров) + ", оплат " + СокрЛП(таблОплаты.КоличествоСтрок()));
		КонецЕсли;
		Если КУДиР_ЭтоСчет62(Опер.Кредит.Счет) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Если КУДиР_ЭтоДенежныйСчет(Опер.Дебет.Счет) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Контр = Опер.Кредит.Контрагенты;
		cid = КУДиР_ИдКонтрагента(Контр);
		Если КУДиР_КлиентВВыборке(cid) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Док = "";
		ДокСсылка = Опер.Документ;
		Если ДокОбъект.НайтиДокумент(ДокСсылка) = 1 Тогда
			Док = ДокОбъект.ТекущийДокумент();
		КонецЕсли;
		Если ПустоеЗначение(Док) = 1 Тогда
			КУДиР_Сбой("проводка оплаты без документа операции, дата=" + КУДиР_ДатаISO(Опер.ДатаОперации) + ", сумма=" + СокрЛП(Опер.Сумма));
			Продолжить;
		КонецЕсли;
		ВидДок = СокрЛП(Док.Вид());
		Kind = КУДиР_ВидОплаты(ВидДок);
		Gran = "POSTING";
		НомПров = КУДиР_НомерПроводкиОпер(Опер);
		Если Kind = "CASH" Тогда
			Gran = "DOCUMENT";
			НомПровПоле = "";
		Иначе
			НомПровПоле = НомПров;
		КонецЕсли;
		payId = КУДиР_ИдОплаты(Kind, Док, Опер);
		Дог = Опер.Кредит.Договоры;
		ВидР = Опер.Кредит.ВидыРасчетовСПокупателями;
		did = КУДиР_ИдДоговора(Контр, Дог);
		Поз = КУДиР_ПозицияДок(Док);
		ДтОпер = Опер.ДатаОперации;
		isT = "0";
		Если (ДтОпер >= датанач) И (ДтОпер <= датакон) Тогда
			isT = "1";
		КонецЕсли;
		Если ПустоеЗначение(Дог) = 0 Тогда
			КУДиР_ЗапомнитьID("CONTRACT", did, Дог, "", КУДиР_НомерДоговора(Дог), КУДиР_ДатаДоговора(Дог), "", "");
		КонецЕсли;
		КУДиР_ЗапомнитьID("PAYMENT", payId, Док, ВидДок, СокрЛП(Док.НомерДок), КУДиР_ДатаISO(Док.ДатаДок), Поз, НомПров);
		Если глОшибка = 1 Тогда
			Продолжить;
		КонецЕсли;
		коп = КУДиР_Копейки(Опер.Сумма);
		Если КУДиР_НайтиОплату(payId) = 1 Тогда
			нСтр = Число(глPaymentsIdx.Item(payId));
			таблОплаты.НомерСтроки = нСтр;
			таблОплаты.СуммаКоп = Число(таблОплаты.СуммаКоп) + коп;
			// Справочные поля payments очищаются при расхождении; CONTRACT каждой корреспонденции уже в id_map.
			Если таблОплаты.ДоговорID <> did Тогда
				таблОплаты.ДоговорID = "";
				таблОплаты.Договор = "";
				таблОплаты.НомерДоговора = "";
				таблОплаты.ДатаДоговора = "";
			КонецЕсли;
			Если таблОплаты.ВидРасчетовID <> КУДиР_ИдВидаРасчетов(ВидР) Тогда
				таблОплаты.ВидРасчетовID = "";
				таблОплаты.ВидРасчетовСПокупателем = "";
			КонецЕсли;
		Иначе
			таблОплаты.НоваяСтрока();
			таблОплаты.payment_id = payId;
			таблОплаты.is_target = isT;
			таблОплаты.PaymentKind = Kind;
			таблОплаты.PaymentGranularity = Gran;
			таблОплаты.ДатаОперации = КУДиР_ДатаISO(ДтОпер);
			таблОплаты.ПозицияДокумента = Поз;
			таблОплаты.НомерПроводкиВДокументе = НомПровПоле;
			таблОплаты.НомерСтрокиВДокументе = "";
			таблОплаты.ВидДокументаОплаты = ВидДок;
			таблОплаты.НомерДокументаОплаты = СокрЛП(Док.НомерДок);
			таблОплаты.ДатаДокументаОплаты = КУДиР_ДатаISO(Док.ДатаДок);
			таблОплаты.СуммаКоп = коп;
			таблОплаты.СчетДебета = СокрЛП("" + Опер.Дебет.Счет);
			таблОплаты.СчетКредита = СокрЛП("" + Опер.Кредит.Счет);
			таблОплаты.КонтрагентID = cid;
			таблОплаты.Контрагент = СокрЛП(Контр);
			таблОплаты.ДоговорID = did;
			таблОплаты.Договор = СокрЛП(Дог);

Процедура КУДиР_СобратьLedger(историяНач, датакон, таблКлиенты)
	КУДиР_ОткрытьCSV("ledger62.csv", "ledger_event_id;source_document_id;payment_id;ДатаОперации;ПозицияДокумента;НомерСтрокиВДокументе;НомерПроводкиВДокументе;ВидДокумента;НомерДокумента;ДатаДокумента;СуммаКоп;СчетДебета;СчетКредита;Счет62;Сторона62;КонтрагентID;Контрагент;ДоговорID;Договор;НомерДоговора;ДатаДоговора;ВидРасчетовID;ВидРасчетовСПокупателем;СодержаниеПроводки;Содержание;ПредставлениеПроводки;КомментарийДокумента;ПервичныйДокумент");
	Опер = СоздатьОбъект("Операция");
	ДокОбъект = СоздатьОбъект("Документ");
	СписокПроводок = "62,*;*,62;";
	Опер.ВыбратьОперацииСПроводками(историяНач, датакон, СписокПроводок);
	нПров = 0;
	Пока Опер.ПолучитьПроводку() = 1 Цикл
		нПров = нПров + 1;
		Если (нПров = 1) Или (Цел(нПров / 50) * 50 = нПров) Тогда
			КУДиР_Состояние("Ledger 62: проводок " + СокрЛП(нПров) + ", строк " + СокрЛП(глСтрокCSV));
		КонецЕсли;
		Если (нПров = 1) Или (Цел(нПров / 500) * 500 = нПров) Тогда
			КУДиР_ЛогФайл(2, "Ledger 62: проводок " + СокрЛП(нПров) + ", строк " + СокрЛП(глСтрокCSV));
		КонецЕсли;
		ЕстьД = КУДиР_ЭтоСчет62(Опер.Дебет.Счет);
		ЕстьК = КУДиР_ЭтоСчет62(Опер.Кредит.Счет);
		Если (ЕстьД = 0) И (ЕстьК = 0) Тогда
			Продолжить;
		КонецЕсли;
		Док = "";
		Если ДокОбъект.НайтиДокумент(Опер.Документ) = 1 Тогда
			Док = ДокОбъект.ТекущийДокумент();
		КонецЕсли;
		Если ПустоеЗначение(Док) = 1 Тогда
			КУДиР_ЛогУр(1, "КУДиР выгрузка: движение 62 без документа операции, дата=" + КУДиР_ДатаISO(Опер.ДатаОперации) + ", сумма=" + СокрЛП(Опер.Сумма) + ". Строка ledger выгружается, source_document_id пустой.");
		КонецЕсли;
		Если ЕстьД = 1 Тогда
			Контр = Опер.Дебет.Контрагенты;
			cid = КУДиР_ИдКонтрагента(Контр);
			Если КУДиР_КлиентВВыборке(cid) = 1 Тогда
				КУДиР_ДобавитьСтрокуLedger(Док, Опер, "D", Контр, Опер.Дебет.Договоры, Опер.Дебет.ВидыРасчетовСПокупателями, Опер.Дебет.Счет);
			КонецЕсли;
		КонецЕсли;
		Если ЕстьК = 1 Тогда
			Контр = Опер.Кредит.Контрагенты;
			cid = КУДиР_ИдКонтрагента(Контр);
			Если КУДиР_КлиентВВыборке(cid) = 1 Тогда
				КУДиР_ДобавитьСтрокуLedger(Док, Опер, "K", Контр, Опер.Кредит.Договоры, Опер.Кредит.ВидыРасчетовСПокупателями, Опер.Кредит.Счет);
			КонецЕсли;
		КонецЕсли;
		Если глОшибка = 1 Тогда
			КУДиР_ЗакрытьCSV();
			Возврат;
		КонецЕсли;
	КонецЦикла;
	КУДиР_Этап("Ledger 62: готово, проводок " + СокрЛП(нПров) + ", строк " + СокрЛП(глСтрокCSV));
	КУДиР_ЗакрытьCSV();
КонецПроцедуры

Процедура КУДиР_ДобавитьДокумент(Док)
	Если ПустоеЗначение(Док) = 1 Тогда
		Возврат;
	КонецЕсли;
	ВидДок = СокрЛП(Док.Вид());
	Role = КУДиР_РольДокумента(ВидДок);
	Если Role = "" Тогда
		Возврат;
	КонецЕсли;
	docId = КУДиР_ИдДокументаПозиция(Док);
	Если docId = "" Тогда
		Возврат;
	КонецЕсли;
	Контр = КУДиР_КонтрагентДокумента(Док);
	Если глОшибка = 1 Тогда
		Возврат;
	КонецЕсли;
	cid = КУДиР_ИдКонтрагента(Контр);
	Если КУДиР_КлиентВВыборке(cid) = 0 Тогда
		Возврат;
	КонецЕсли;
	Поз = КУДиР_ПозицияДок(Док);
	Если КУДиР_ЕстьКлюч(глDocuments, docId) = 1 Тогда
		КУДиР_ЗапомнитьID("DOCUMENT", docId, Док, ВидДок, СокрЛП(Док.НомерДок), КУДиР_ДатаISO(Док.ДатаДок), Поз, "");
		Возврат;
	КонецЕсли;
	Дог = КУДиР_РеквизитДокумента(Док, "Договор");
	did = КУДиР_ИдДоговора(Контр, Дог);
	ВидР = "";
	Если Role = "RECEIVABLE" Тогда
		ВидР = КУДиР_ВидРасчетовРеквизитДок(Док);
		Если ПустоеЗначение(ВидР) = 1 Тогда
			Если КУДиР_ПроведенДок(Док) = "1" Тогда
				глВидРасчетовНесколько = 0;
				ВидР = КУДиР_ВидРасчетовИзПроводок62(Док);
				Если (ПустоеЗначение(ВидР) = 1) И (глВидРасчетовНесколько <> 1) Тогда
					КУДиР_Сбой("у проведённой реализации " + ВидДок + " №" + СокрЛП(Док.НомерДок) + " нет вида расчётов ни в реквизите документа, ни в проводке 62.");
					Возврат;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;
	СуммаКоп = КУДиР_СуммаДокументаКоп(Док);
	Если глОшибка = 1 Тогда
		Возврат;
	КонецЕсли;
	СуммаОблагаемаяКоп = 0;
	СуммаНеоблагаемаяКоп = 0;
	СуммаНДСКоп = 0;
	Если КУДиР_ЕстьРеквизитДокумента(ВидДок, "НДС") = 0 Тогда
		КУДиР_Сбой("нет колонки НДС у вида документа " + ВидДок);
		Возврат;
	КонецЕсли;
	Док.ВыбратьСтроки();
	нСтр = 0;
	отрицНДС = 0;
	Пока Док.ПолучитьСтроку() = 1 Цикл
		нСтр = нСтр + 1;
		сумСтр = КУДиР_Копейки(Док.Сумма);
		ндсСтр = КУДиР_Копейки(Док.НДС);
		Если ндсСтр > 0 Тогда
			СуммаОблагаемаяКоп = СуммаОблагаемаяКоп + сумСтр;
			СуммаНДСКоп = СуммаНДСКоп + ндсСтр;
		ИначеЕсли ндсСтр = 0 Тогда
			СуммаНеоблагаемаяКоп = СуммаНеоблагаемаяКоп + сумСтр;
		Иначе
			отрицНДС = 1;
			КУДиР_ЛогУр(1, "КУДиР выгрузка: отрицательный НДС строки табличной части " + ВидДок + " №" + СокрЛП(Док.НомерДок) + ". Агрегаты не утверждаются как налоговый факт.");
		КонецЕсли;
	КонецЦикла;
	Если (нСтр = 0) И (Число(СуммаКоп) <> 0) Тогда
		СуммаОблагаемаяКоп = 0;
		СуммаНеоблагаемаяКоп = 0;
		СуммаНДСКоп = 0;
		КУДиР_ЛогУр(1, "КУДиР выгрузка: пустая табличная часть при ненулевой сумме, " + ВидДок + " №" + СокрЛП(Док.НомерДок) + ". Агрегаты НДС = 0.");
	ИначеЕсли (нСтр > 0) И (отрицНДС = 0) Тогда
		Если (СуммаОблагаемаяКоп + СуммаНеоблагаемаяКоп) <> Число(СуммаКоп) Тогда
			КУДиР_ЛогУр(1, "КУДиР выгрузка: Облагаемая+Необлагаемая <> СуммаКоп, " + ВидДок + " №" + СокрЛП(Док.НомерДок) + ".");
		КонецЕсли;
	КонецЕсли;
	КУДиР_ЗапомнитьID("DOCUMENT", docId, Док, ВидДок, СокрЛП(Док.НомерДок), КУДиР_ДатаISO(Док.ДатаДок), Поз, "");
	Если глОшибка = 1 Тогда
		Возврат;
	КонецЕсли;
	глDocuments.Add(docId, 1);
	Если ПустоеЗначение(Дог) = 0 Тогда
		КУДиР_ЗапомнитьID("CONTRACT", did, Дог, "", КУДиР_НомерДоговора(Дог), КУДиР_ДатаДоговора(Дог), "", "");
	КонецЕсли;
	vrId = КУДиР_ИдВидаРасчетов(ВидР);
	ТекстСтроки = КУДиР_ПолеCSV(docId) + ";"
		+ КУДиР_ПолеCSV(Role) + ";"
		+ КУДиР_ПолеCSV(КУДиР_ТипДокумента(ВидДок, Role)) + ";"
		+ КУДиР_ПолеCSV(ВидДок) + ";"
		+ КУДиР_ПолеCSV(СокрЛП(Док.НомерДок)) + ";"
		+ КУДиР_ПолеCSV(КУДиР_ДатаISO(Док.ДатаДок)) + ";"
		+ КУДиР_ПолеCSV(Поз) + ";"
		+ КУДиР_ПолеCSV(cid) + ";"
		+ КУДиР_ПолеCSV(СокрЛП(Контр)) + ";"
		+ КУДиР_ПолеCSV(did) + ";"
		+ КУДиР_ПолеCSV(СокрЛП(Дог)) + ";"
		+ КУДиР_ПолеCSV(КУДиР_НомерДоговора(Дог)) + ";"
		+ КУДиР_ПолеCSV(КУДиР_ДатаДоговора(Дог)) + ";"
		+ КУДиР_ПолеCSV(vrId) + ";"
		+ КУДиР_ПолеCSV(СокрЛП(ВидР)) + ";"
		+ КУДиР_ПолеCSV(СуммаКоп) + ";"
		+ КУДиР_ПолеCSV(СуммаОблагаемаяКоп) + ";"
		+ КУДиР_ПолеCSV(СуммаНеоблагаемаяКоп) + ";"
		+ КУДиР_ПолеCSV(СуммаНДСКоп) + ";"
		+ КУДиР_ПолеCSV(КУДиР_ПроведенДок(Док)) + ";"
		+ КУДиР_ПолеCSV(КУДиР_ПомеченДок(Док)) + ";"
		+ КУДиР_ПолеCSV(КУДиР_КомментарийДок(Док)) + ";"
		+ КУДиР_ПолеCSV("");
	КУДиР_ПисатьCSVСтроку(ТекстСтроки);

Функция КУДиР_ВидРасчетовИзПроводок62(Док)
	глВидРасчетовНесколько = 0;
	Если ПустоеЗначение(Док) = 1 Тогда
		Возврат "";
	КонецЕсли;
	Опер = СоздатьОбъект("Операция");
	Если Опер.НайтиОперацию(Док) = 0 Тогда
		Возврат "";
	КонецЕсли;
	ВидР = "";
	ВидРИд = "";
	Разных = 0;
	НПров = Опер.КоличествоПроводок();
	н = 1;
	Пока н <= НПров Цикл
		Опер.ПолучитьПроводку(н);
		Тек = "";
		Если КУДиР_ЭтоСчет62(Опер.Дебет.Счет) = 1 Тогда
			Тек = Опер.Дебет.ВидыРасчетовСПокупателями;
		ИначеЕсли КУДиР_ЭтоСчет62(Опер.Кредит.Счет) = 1 Тогда
			Тек = Опер.Кредит.ВидыРасчетовСПокупателями;
		КонецЕсли;
		ТекИд = КУДиР_ИдВидаРасчетов(Тек);
		Если ТекИд <> "" Тогда
			Если ВидРИд = "" Тогда
				ВидРИд = ТекИд;
				ВидР = Тек;
			ИначеЕсли ВидРИд <> ТекИд Тогда
				Разных = 1;
			КонецЕсли;
		КонецЕсли;
		н = н + 1;
	КонецЦикла;
	Если Разных = 1 Тогда
		глВидРасчетовНесколько = 1;
		КУДиР_ЛогУр(1, "КУДиР выгрузка: документ " + СокрЛП(Док.Вид()) + " №" + СокрЛП(Док.НомерДок) + " имеет несколько видов расчётов в проводках 62, ВидРасчетовID документа оставлен пустым (аналитика в ledger62).");
		Возврат "";
	КонецЕсли;
	Возврат ВидР;

Процедура КУДиР_ОбойтиВидДокумента(Знач ИмяВида, историяНач, горизонт)
	Если КУДиР_ЕстьВидДокумента(ИмяВида) = 0 Тогда
		КУДиР_ЛогУр(1, "в конфигурации отсутствует вид документа " + ИмяВида);
		Возврат;
	КонецЕсли;
	КУДиР_Этап("Документы " + ИмяВида + ": обход");
	Док = СоздатьОбъект("Документ." + ИмяВида);
	Док.ВыбратьДокументы(историяНач, горизонт);
	н = 0;
	Пока Док.ПолучитьДокумент() = 1 Цикл
		н = н + 1;
		Если (н = 1) Или (Цел(н / 50) * 50 = н) Тогда

Функция КУДиР_РазобратьCSVСтроку(Стр)
	сп = СоздатьОбъект("СписокЗначений");
	Поле = "";
	ВКавычках = 0;
	Кавычка = Симв(34);
	й = 1;

		Сим = Сред(Стр, й, 1);
		Если ВКавычках = 1 Тогда
			Если Сим = Кавычка Тогда
				Если й < Длн Тогда
					Если Сред(Стр, й + 1, 1) = Кавычка Тогда
						Поле = Поле + Кавычка;
						й = й + 1;
					Иначе
						ВКавычках = 0;
					КонецЕсли;
				Иначе
					ВКавычках = 0;
				КонецЕсли;
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		Иначе
			Если Сим = Кавычка Тогда
				ВКавычках = 1;
			ИначеЕсли Сим = ";" Тогда
				сп.ДобавитьЗначение(Поле);
				Поле = "";
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		КонецЕсли;
		й = й + 1;
	КонецЦикла;
	Если ВКавычках = 1 Тогда
		КУДиР_Сбой("незакрытая кавычка CSV-записи.");
	КонецЕсли;
	сп.ДобавитьЗначение(Поле);
	Возврат сп;

Функция КУДиР_ПрочитатьЛогическуюCSVСтроку(TS)
	// Логическая CSV-запись: переносы внутри кавычек, удвоенные кавычки.
	// EOF внутри кавычек = FAILED. Возврат 0 - конец потока или ошибка.
	сп = СоздатьОбъект("СписокЗначений");
	Поле = "";
	ВКавычках = 0;
	Кавычка = Симв(34);
	БылиДанные = 0;
	Если (TS.AtEndOfStream = 1) И (глCSVБуфер = "") Тогда
		Возврат 0;
	КонецЕсли;
	Пока 1 = 1 Цикл
		Сим = КУДиР_CSVСледСимвол(TS);
		Если Сим = "" Тогда
			Если ВКавычках = 1 Тогда
				КУДиР_Сбой("EOF внутри кавычек логической CSV-записи.");
				Возврат 0;
			КонецЕсли;
			Если (БылиДанные = 0) И (Поле = "") И (сп.РазмерСписка() = 0) Тогда
				Возврат 0;
			КонецЕсли;
			сп.ДобавитьЗначение(Поле);
			Возврат сп;
		КонецЕсли;
		БылиДанные = 1;
		Если ВКавычках = 1 Тогда
			Если Сим = Кавычка Тогда
				След = КУДиР_CSVСледСимвол(TS);
				Если След = Кавычка Тогда
					Поле = Поле + Кавычка;
				Иначе
					ВКавычках = 0;
					Если След <> "" Тогда
						КУДиР_CSVВернутьСимвол(След);
					КонецЕсли;
				КонецЕсли;
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		Иначе
			Если Сим = Кавычка Тогда
				ВКавычках = 1;
			ИначеЕсли Сим = ";" Тогда
				сп.ДобавитьЗначение(Поле);
				Поле = "";
			ИначеЕсли (Сим = Симв(13)) Или (Сим = Симв(10)) Тогда
				Если Сим = Симв(13) Тогда
					След = КУДиР_CSVСледСимвол(TS);
					Если След <> Симв(10) Тогда
						Если След <> "" Тогда
							КУДиР_CSVВернутьСимвол(След);
						КонецЕсли;
					КонецЕсли;
				КонецЕсли;
				сп.ДобавитьЗначение(Поле);
				Возврат сп;
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		КонецЕсли;
	КонецЦикла;
	Возврат 0;

Функция КУДиР4_СледСимвол(TS)
	// Чтение блоками по 8192 символа; позиция в буфере позволяет вернуть один прочитанный символ.
	Если гл4БуферПоз > СтрДлина(гл4Буфер) Тогда
		гл4Буфер = "";
		гл4БуферПоз = 1;
		Если TS.AtEndOfStream <> 0 Тогда
			Возврат "";
		КонецЕсли;
		гл4Буфер = TS.Read(8192);
	КонецЕсли;
	Если гл4БуферПоз > СтрДлина(гл4Буфер) Тогда
		Возврат "";
	КонецЕсли;
	Сим = Сред(гл4Буфер, гл4БуферПоз, 1);
	гл4БуферПоз = гл4БуферПоз + 1;
	Возврат Сим;

Функция КУДиР4_ПрочитатьЗапись(TS, Знач ИмяФайла)
	// Логическая CSV-запись (№1 §30, №4 §5): ";" - разделитель, кавычка в поле удваивается, переносы
	// внутри кавычек сохраняются как есть. Запись завершается CR LF. Символы различаются по кодам, без
	// сравнения строк 1С. Возврат: СписокЗначений или 0 в конце файла.
	сп = СоздатьОбъект("СписокЗначений");
	Поле = "";
	ВКавычках = 0;
	ПослеКавычки = 0;
	БылиДанные = 0;
	Пока 1 = 1 Цикл
		Сим = КУДиР4_СледСимвол(TS);
		Если СтрДлина(Сим) = 0 Тогда
			Если ВКавычках = 1 Тогда
				КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "конец файла внутри кавычек");
				Возврат 0;
			КонецЕсли;
			Если БылиДанные = 0 Тогда
				Возврат 0;
			КонецЕсли;
			КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "последняя запись не завершена CR LF");
			сп.ДобавитьЗначение(Поле);
			Возврат сп;
		КонецЕсли;
		БылиДанные = 1;
		Код = КодСимв(Сим);
		Если ВКавычках = 1 Тогда
			Если Код = 34 Тогда
				ВКавычках = 0;
				ПослеКавычки = 1;
				След = КУДиР4_СледСимвол(TS);
				Если СтрДлина(След) > 0 Тогда
					Если КодСимв(След) = 34 Тогда
						Поле = Поле + Сим;
						ВКавычках = 1;
						ПослеКавычки = 0;
					Иначе
						гл4БуферПоз = гл4БуферПоз - 1;
					КонецЕсли;
				КонецЕсли;
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		ИначеЕсли Код = 59 Тогда
			сп.ДобавитьЗначение(Поле);
			Поле = "";
			ПослеКавычки = 0;
		ИначеЕсли (Код = 13) Или (Код = 10) Тогда
			Если Код = 13 Тогда
				След = КУДиР4_СледСимвол(TS);
				Если СтрДлина(След) = 0 Тогда
					КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "запись завершена одиночным CR в конце файла");
				ИначеЕсли КодСимв(След) <> 10 Тогда
					гл4БуферПоз = гл4БуферПоз - 1;
					КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "запись завершена одиночным CR");
				КонецЕсли;
			Иначе
				КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "запись завершена LF без CR");
			КонецЕсли;
			сп.ДобавитьЗначение(Поле);
			Возврат сп;
		ИначеЕсли ПослеКавычки = 1 Тогда
			КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "символ после закрывающей кавычки, код " + СокрЛП(Код));
			Поле = Поле + Сим;
		ИначеЕсли Код = 34 Тогда
			Если СтрДлина(Поле) = 0 Тогда
				ВКавычках = 1;
			Иначе
				КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "кавычка внутри поля без кавычек");
				Поле = Поле + Сим;
			КонецЕсли;
		Иначе
			Поле = Поле + Сим;
		КонецЕсли;
	КонецЦикла;
	Возврат 0;

Функция КУДиР4_ОткрытьЧтение(Знач ИмяФайла, Знач Заголовок)
	// Чтение уже закрытого файла каталога запуска; заголовок сверяется точно.
	FSO = СоздатьОбъект("Scripting.FileSystemObject");
	TS = FSO.OpenTextFile(гл4Каталог + "\" + ИмяФайла, 1, 0);
	гл4Буфер = "";
	гл4БуферПоз = 1;
	гл4ЗаголовокЧтения = КУДиР4_РазобратьЗаголовок(Заголовок);
	Заг = КУДиР4_ПрочитатьЗапись(TS, ИмяФайла);
	Если ТипЗначенияСтр(Заг) <> "СписокЗначений" Тогда
		КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "нет заголовка");
	ИначеЕсли КУДиР_СтрокиРавны(КУДиР4_СтрокаЗаписи(Заг), Заголовок) = 0 Тогда
		КУДиР4_Ошибка(ИмяФайла, "PACKAGE", "", "CSV_INVALID", "заголовок отличается: [" + КУДиР4_СтрокаЗаписи(Заг) + "]");

Процедура КУДиР4_СверитьЗапись(сп, TS, Знач ИмяФайла, Знач КодДиаг)
	// Повторное чтение: значения, вычисленные заново, сравниваются с прочитанной CSV-записью поле за полем.
	КлючЗаписи = "" + сп.ПолучитьЗначение(1);
	Факт = КУДиР4_ПрочитатьЗапись(TS, ИмяФайла);
	Если ТипЗначенияСтр(Факт) <> "СписокЗначений" Тогда
		гл4Расхождений = гл4Расхождений + 1;
		КУДиР4_Ошибка(ИмяФайла, КлючЗаписи, "", КодДиаг, "при повторном чтении получена запись, которой нет в файле");
		Возврат;
	КонецЕсли;
	Если Факт.РазмерСписка() <> сп.РазмерСписка() Тогда
		гл4Расхождений = гл4Расхождений + 1;
		КУДиР4_Ошибка(ИмяФайла, КлючЗаписи, "", "CSV_INVALID", "полей в записи " + СокрЛП(Факт.РазмерСписка()) + ", ожидалось " + СокрЛП(сп.РазмерСписка()));
		Возврат;
	КонецЕсли;
	н = 1;
	Пока н <= сп.РазмерСписка() Цикл
		Если КУДиР_СтрокиРавны(сп.ПолучитьЗначение(н), Факт.ПолучитьЗначение(н)) = 0 Тогда
			гл4Расхождений = гл4Расхождений + 1;
			КУДиР4_Ошибка(ИмяФайла, КлючЗаписи, "" + гл4ЗаголовокЧтения.ПолучитьЗначение(н), КодДиаг, "вычислено [" + сп.ПолучитьЗначение(н) + "], в файле [" + Факт.ПолучитьЗначение(н) + "]");
		КонецЕсли;
		н = н + 1;
	КонецЦикла;

Процедура КУДиР4_СторонаПроводки(Опер, Знач Сторона, Знач Ключ, Знач Сист, Знач Поз, Знач НомПров, Знач ВидДок, Знач НомДок, Знач ДатаДок, Знач ДатаОпер, Знач Коммент, Знач Проход, TS)
	// Одна сторона 62 корреспонденции = одна строка. Проход 1 пишет строку и регистрирует ID,
	// проход 2 вычисляет её заново и сверяет с записанным файлом.
	Если Сторона = "D" Тогда
		Счет62 = СокрЛП("" + Опер.Дебет.Счет);
		Денежная = КУДиР_ЭтоДенежныйСчет(Опер.Кредит.Счет);
	Иначе
		Счет62 = СокрЛП("" + Опер.Кредит.Счет);
		Денежная = КУДиР_ЭтоДенежныйСчет(Опер.Дебет.Счет);
	КонецЕсли;
	ИдДок = "";
	ИдСтроки = "";
	НомерСтороны = "";
	Если СтрДлина(Ключ) > 0 Тогда
		ИдДок = "DOC|" + Ключ;
		Если СтрДлина(НомПров) > 0 Тогда
			НомерСтороны = СокрЛП(КУДиР4_СледНомерСтороны(Ключ, НомПров));
			ИдСтроки = "LED|" + Ключ + "|" + НомПров + "|" + НомерСтороны;
		КонецЕсли;
	КонецЕсли;
	// Аналитика своей стороны 62 (№1 §11), не копируется с другой стороны переноса.
	ИдКонтр = "";
	ИдДог = "";
	Дог = "";
	Попытка
		Если Сторона = "D" Тогда
			Контр = Опер.Дебет.Контрагенты;
			Дог = Опер.Дебет.Договоры;
		Иначе
			Контр = Опер.Кредит.Контрагенты;
			Дог = Опер.Кредит.Договоры;
		КонецЕсли;
		ИдКонтр = КУДиР_ИдКонтрагента(Контр);
		Если СтрДлина(ИдКонтр) > 0 Тогда
			ИдДог = КУДиР_ИдДоговора(Контр, Дог);
		КонецЕсли;
	Исключение
		Причина = ОписаниеОшибки();
		ИдКонтр = "";
		ИдДог = "";
		Если Проход = 1 Тогда
			Если КУДиР4_Впервые("АНАЛИТИКА|" + Счет62) = 1 Тогда
				КУДиР4_Предупреждение("e02_ids.csv", ИдСтроки, "КонтрагентID", "ANALYTICS_UNKNOWN", "покупатель/договор стороны " + Счет62 + " не читаются: " + Причина + " (сообщение одно на счёт)");
			КонецЕсли;
		КонецЕсли;
	КонецПопытки;
	Если Проход = 1 Тогда
		Если СтрДлина(ИдКонтр) = 0 Тогда
			Если КУДиР4_Впервые("ПОКУПАТЕЛЬ|" + Счет62) = 1 Тогда
				КУДиР4_Предупреждение("e02_ids.csv", ИдСтроки, "КонтрагентID", "ANALYTICS_UNKNOWN", "пустой покупатель стороны " + Счет62 + "; строки сохранены (сообщение одно на счёт)");
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;
	ИдВида = "";
	Попытка
		Если Сторона = "D" Тогда
			ВидР = Опер.Дебет.ВидыРасчетовСПокупателями;
		Иначе
			ВидР = Опер.Кредит.ВидыРасчетовСПокупателями;
		КонецЕсли;
		ИдВида = КУДиР4_ИдВидаРасчетов(ВидР, Проход, ИдСтроки);
	Исключение
		Причина = ОписаниеОшибки();
		ИдВида = "";
		Если Проход = 1 Тогда
			Если КУДиР4_Впервые("ВИД|" + Счет62) = 1 Тогда
				КУДиР4_Предупреждение("e02_ids.csv", ИдСтроки, "ВидРасчетовID", "ANALYTICS_UNKNOWN", "вид расчётов стороны " + Счет62 + " не читается: " + Причина + " (сообщение одно на счёт; для 62.2 отсутствие вида штатно)");
			КонецЕсли;
		КонецЕсли;
	КонецПопытки;
	Если Проход = 1 Тогда
		Если СтрДлина(ИдДог) > 0 Тогда
			КУДиР4_ЗапомнитьДоговор(Дог, ИдДог);
		КонецЕсли;
	КонецЕсли;
	// payment_id: денежная корреспонденция 62 с 50/51/55 и вид документа из подтверждённого перечня.
	ВидОпл = "";
	ИдОпл = "";
	Если Денежная = 1 Тогда
		Если СтрДлина(Ключ) > 0 Тогда
			ВидОпл = КУДиР4_ВидОплаты(ВидДок);
			Если СтрДлина(ВидОпл) = 0 Тогда
				Если Проход = 1 Тогда
					Если КУДиР4_Впервые("ВИДОПЛАТЫ|" + ВидДок) = 1 Тогда
						КУДиР4_Предупреждение("e02_ids.csv", ИдСтроки, "payment_id", "PAYMENT_MEMBERSHIP_UNKNOWN", "денежная корреспонденция 62 в документе вида " + ВидДок + " вне перечня №1 §14; payment_id пуст (сообщение одно на вид)");
					КонецЕсли;
				КонецЕсли;
			ИначеЕсли ВидОпл = "CASH" Тогда
				ИдОпл = КУДиР4_ИдОплаты(ВидОпл, Ключ, НомПров);
			ИначеЕсли СтрДлина(НомПров) > 0 Тогда
				ИдОпл = КУДиР4_ИдОплаты(ВидОпл, Ключ, НомПров);
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;
	ВидОплПоле = "";
	Если СтрДлина(ИдОпл) > 0 Тогда
		ВидОплПоле = ВидОпл;
		Если Проход = 1 Тогда
			НомПровКарты = "";
			Если ВидОпл <> "CASH" Тогда
				НомПровКарты = НомПров;
			КонецЕсли;
			КУДиР4_ЗапомнитьИд("PAYMENT", ИдОпл, Сист, ВидДок, НомДок, ДатаДок, Поз, НомПровКарты);
			КУДиР4_Учесть("сторон 62 с payment_id " + ВидОпл);
		КонецЕсли;
	КонецЕсли;
	сп = СоздатьОбъект("СписокЗначений");
	сп.ДобавитьЗначение(ИдСтроки);
	сп.ДобавитьЗначение(ИдДок);
	сп.ДобавитьЗначение(ИдОпл);
	сп.ДобавитьЗначение(ВидОплПоле);
	сп.ДобавитьЗначение(ДатаОпер);
	сп.ДобавитьЗначение(Поз);
	сп.ДобавитьЗначение(НомПров);
	сп.ДобавитьЗначение(НомерСтороны);
	сп.ДобавитьЗначение(Сторона);
	сп.ДобавитьЗначение(Счет62);
	сп.ДобавитьЗначение(СокрЛП("" + Опер.Дебет.Счет));
	сп.ДобавитьЗначение(СокрЛП("" + Опер.Кредит.Счет));
	сп.ДобавитьЗначение("" + КУДиР_Копейки(Опер.Сумма));
	сп.ДобавитьЗначение(ВидДок);
	сп.ДобавитьЗначение(НомДок);
	сп.ДобавитьЗначение(ДатаДок);
	сп.ДобавитьЗначение(ИдКонтр);
	сп.ДобавитьЗначение(ИдДог);
	сп.ДобавитьЗначение(ИдВида);
	сп.ДобавитьЗначение(КУДиР4_ТекстПроводки(Опер, "СодержаниеПроводки", Проход));
	сп.ДобавитьЗначение(КУДиР4_ТекстПроводки(Опер, "Содержание", Проход));
	сп.ДобавитьЗначение(Коммент);
	Если Проход = 1 Тогда
		Если СтрДлина(ИдСтроки) > 0 Тогда
			Если гл4ИдСтрок.Exists(ИдСтроки) <> 0 Тогда
				КУДиР4_Ошибка("e02_ids.csv", ИдСтроки, "ledger_event_id", "ID_COLLISION", "повторный ledger_event_id в одном наборе");
			Иначе
				гл4ИдСтрок.Add(ИдСтроки, 1);
			КонецЕсли;
		КонецЕсли;
		КУДиР4_ПисатьДанные(КУДиР4_СтрокаЗаписи(сп));
		КУДиР4_Учесть("сторон 62 на счёте " + Счет62);
	Иначе
		КУДиР4_СверитьЗапись(сп, TS, "e02_ids.csv", "READ_ERROR");
	КонецЕсли;

Процедура КУДиР4_ОбойтиПроводки(датанач, датакон, Знач Проход, TS)
	// Все проводки окна без строки-фильтра; стороны 62 отбираются кодом. Документ проводки - только
	// НайтиДокумент(Опер.Документ) -> ТекущийДокумент() (КП0.1).
	Опер = СоздатьОбъект("Операция");
	ДокОбъект = СоздатьОбъект("Документ");
	гл4ПослКлюч = "";
	гл4ПослПроводка = "";
	гл4НомерСтороны = 0;
	Осмотрено = 0;
	С62 = 0;
	Опер.ВыбратьОперацииСПроводками(датанач, датакон);
	Пока Опер.ПолучитьПроводку() = 1 Цикл
		Осмотрено = Осмотрено + 1;
		Если (Осмотрено = 1) Или (Цел(Осмотрено / 100) * 100 = Осмотрено) Тогда
			Состояние("Э0.2 " + гл4Этап + ": проводок " + СокрЛП(Осмотрено) + ", с 62: " + СокрЛП(С62) + ", ошибок " + СокрЛП(гл4Ошибок));
		КонецЕсли;
		Если гл4Фатально = 1 Тогда
			Прервать;
		КонецЕсли;
		Если гл4Ошибок >= 200 Тогда
			гл4Фатально = 1;
			КУДиР_ЛогУр(0, "Э0.2: 200 ошибок, обход проводок остановлен");
			Прервать;
		КонецЕсли;
		ЕстьД = КУДиР_ЭтоСчет62(Опер.Дебет.Счет);
		ЕстьК = КУДиР_ЭтоСчет62(Опер.Кредит.Счет);
		Если (ЕстьД = 0) И (ЕстьК = 0) Тогда
			Продолжить;
		КонецЕсли;
		С62 = С62 + 1;
		ДатаОпер = КУДиР_ДатаISO(Опер.ДатаОперации);
		Ном = Опер.НомерПроводки();
		НомПров = "";
		Если Ном > 0 Тогда
			Если Ном = Цел(Ном) Тогда
				НомПров = СокрЛП(Ном);
			КонецЕсли;
		КонецЕсли;
		Ссылка = "";
		Если ДокОбъект.НайтиДокумент(Опер.Документ) = 1 Тогда
			Ссылка = ДокОбъект.ТекущийДокумент();
		КонецЕсли;
		Сист = "";
		Ключ = "";
		Поз = "";
		ВидДок = "";
		НомДок = "";
		ДатаДок = "";
		Коммент = "";
		Если ПустоеЗначение(Ссылка) = 0 Тогда
			Сист = ЗначениеВСтрокуВнутр(Ссылка);
			Ключ = КУДиР_КодироватьКлюч(Сист);
			Поз = КУДиР_ПозицияДокТочная(Ссылка);
			ВидДок = СокрЛП(Ссылка.Вид());
			НомДок = СокрЛП(Ссылка.НомерДок);
			ДатаДок = КУДиР_ДатаISO(Ссылка.ДатаДок);
			Коммент = КУДиР4_КомментарийДок(Ссылка);
		КонецЕсли;
		Если Проход = 1 Тогда
			КУДиР4_ПроверитьДокументПроводки(Ключ, Сист, Поз, ВидДок, НомДок, ДатаДок, НомПров, ДатаОпер, Ном);
		КонецЕсли;
		Если ЕстьД = 1 Тогда
			КУДиР4_СторонаПроводки(Опер, "D", Ключ, Сист, Поз, НомПров, ВидДок, НомДок, ДатаДок, ДатаОпер, Коммент, Проход, TS);
		КонецЕсли;
		Если ЕстьК = 1 Тогда
			КУДиР4_СторонаПроводки(Опер, "K", Ключ, Сист, Поз, НомПров, ВидДок, НомДок, ДатаДок, ДатаОпер, Коммент, Проход, TS);
		КонецЕсли;
	КонецЦикла;
	Если Проход = 1 Тогда
		гл4Проводок62 = С62;
	КонецЕсли;
	КУДиР_Лог("Э0.2 " + гл4Этап + ": проводок осмотрено " + СокрЛП(Осмотрено) + ", с 62: " + СокрЛП(С62));
КонецПроцедуры

Процедура КУДиР4_ОбойтиВид(Знач ИмяВида, датанач, датакон, Знач Проход, TS)
	// Наличие вида проверяется созданием объекта, без опроса метаданных.
	Создана = 0;
	Попытка
		Выборка = СоздатьОбъект("Документ." + ИмяВида);
		Создана = 1;
	Исключение
		Причина = ОписаниеОшибки();
		Если Проход = 1 Тогда
			КУДиР4_Предупреждение("e02_docs.csv", "PACKAGE", "ВидДокумента", "SOURCE_FIELD_UNAVAILABLE", "вид документа " + ИмяВида + " не создаётся: " + Причина);
		КонецЕсли;
	КонецПопытки;
	Если Создана = 0 Тогда
		Возврат;
	КонецЕсли;
	Выборка.ВыбратьДокументы(датанач, датакон);
	н = 0;
	Пока Выборка.ПолучитьДокумент() = 1 Цикл
		н = н + 1;
		Если (н = 1) Или (Цел(н / 100) * 100 = н) Тогда
			Состояние("Э0.2 " + гл4Этап + ": " + ИмяВида + " " + СокрЛП(н) + ", ошибок " + СокрЛП(гл4Ошибок));
		КонецЕсли;
		Если гл4Фатально = 1 Тогда
			Прервать;
		КонецЕсли;
		Если гл4Ошибок >= 200 Тогда
			гл4Фатально = 1;
			КУДиР_ЛогУр(0, "Э0.2: 200 ошибок, обход документов остановлен");
			Прервать;
		КонецЕсли;
		// Объект выборки для K не годится (КП0.1: номер 0) - только значение ТекущийДокумент().
		Ссылка = Выборка.ТекущийДокумент();
		Сист = ЗначениеВСтрокуВнутр(Ссылка);
		Ключ = КУДиР_КодироватьКлюч(Сист);
		Поз = КУДиР_ПозицияДокТочная(Ссылка);
		ВидДок = СокрЛП(Ссылка.Вид());
		НомДок = СокрЛП(Ссылка.НомерДок);
		ДатаДок = КУДиР_ДатаISO(Ссылка.ДатаДок);
		ИдДок = "";
		Если СтрДлина(Сист) > 0 Тогда
			ИдДок = "DOC|" + Ключ;
		КонецЕсли;
		Если Проход = 1 Тогда
			Если СтрДлина(Сист) = 0 Тогда
				КУДиР4_Ошибка("e02_docs.csv", ИмяВида + " " + НомДок + " " + ДатаДок, "document_id", "READ_ERROR", "пустая системная ссылка ТекущийДокумент()");
			Иначе
				Если СтрДлина(Поз) = 0 Тогда
					КУДиР4_Ошибка("e02_docs.csv", ИдДок, "ПозицияДокумента", "READ_ERROR", "пустая позиция документа");
				КонецЕсли;
				КУДиР4_ЗапомнитьПозицию(Ключ, Поз, ИдДок);
				КУДиР4_ЗапомнитьИд("DOCUMENT", ИдДок, Сист, ВидДок, НомДок, ДатаДок, Поз, "");
			КонецЕсли;
		КонецЕсли;
		сп = СоздатьОбъект("СписокЗначений");
