Функция КУДиР_Выполнить(датанач, датакон)
	КУДиР_ИнициализироватьПути();
	глТаблID = КУДиР_СоздатьТаблицуID();
	глОшибкаID = 0;
	глМесяцевИстории = 3;
	глМесяцевГоризонта = 3;
	глПослПозиция = "";
	глПослПроводка = "";
	глНомКорр = 0;

	историяНач = ДобавитьМесяц(КУДиР_НачалоГода(датанач), -глМесяцевИстории);
	квКонец = КУДиР_КонецКвартала(датакон);
	горизонт = КУДиР_КонецКвартала(ДобавитьМесяц(датакон, глМесяцевГоризонта));

	таблОплаты = КУДиР_СоздатьТаблицуОплат();
	таблLedger = КУДиР_СоздатьТаблицуLedger();
	таблДокументы = КУДиР_СоздатьТаблицуДокументов();
	таблСальдо = КУДиР_СоздатьТаблицуСальдо();
	таблКлиенты = КУДиР_СоздатьТаблицуКлиентов();

	КУДиР_СобратьОплаты(историяНач, датанач, датакон, таблОплаты, таблКлиенты);
	Если таблКлиенты.КоличествоСтрок() = 0 Тогда
		Сообщить("КУДиР выгрузка: нет целевых оплат 50/51/55 -> 62 за ДатаНач...ДатаКон.");
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;

	КУДиР_СобратьLedger(историяНач, датакон, таблОплаты, таблКлиенты, таблLedger);
	КУДиР_СобратьДокументы(историяНач, горизонт, таблКлиенты, таблДокументы);
	КУДиР_СобратьСальдо(историяНач, таблКлиенты, таблСальдо);

	Если глОшибкаID = 1 Тогда
		Сообщить("КУДиР выгрузка FAILED: коллизия ID -> объект 1С.");
		Возврат СформироватьТаблицуДоходовЧистовая();
	КонецЕсли;

	run_id = КУДиР_НовыйRunId(датанач);
	FSO = СоздатьОбъект("Scripting.FileSystemObject");
	глКаталогОбмена = глКаталогПроекта + "\exchange\" + run_id;
	Если FSO.FolderExists(глКаталогОбмена) = 0 Тогда
		FSO.CreateFolder(глКаталогОбмена);
	КонецЕсли;

	КУДиР_ЗаписатьВыгрузку(run_id, историяНач, датанач, датакон, квКонец, горизонт, таблОплаты, таблLedger, таблДокументы, таблСальдо);

	Код = КУДиР_ЗапуститьPython();
	Если Код <> 0 Тогда
		Сообщить("КУДиР: Python вернул код " + СокрЛП(Код) + ". Смотрите " + глКаталогОбмена + "\\run_status.csv и python_run.log");
	КонецЕсли;

	Возврат КУДиР_ЗагрузитьИВосстановить();
КонецФункции

Процедура КУДиР_ЗапомнитьID(Знач ТипСущ, Знач Ид, Знач Ссылка, Знач Вид, Знач Номер, Знач ДатаСтр, Знач Позиция, Знач НомПров)
	Если СокрЛП(Ид) = "" Тогда
		Возврат;
	КонецЕсли;
	Уже = КУДиР_НайтиСсылку(ТипСущ, Ид);
	Если Уже <> 0 Тогда
		Если Уже <> Ссылка Тогда
			глОшибкаID = 1;
			Сообщить("Коллизия ID: " + ТипСущ + " " + Ид + " указывает на два объекта.");
		КонецЕсли;
		Возврат;
	КонецЕсли;
	ЧужойИд = КУДиР_НайтиИдПоСсылке(ТипСущ, Ссылка);
	Если СокрЛП(ЧужойИд) <> "" Тогда
		Если ЧужойИд <> Ид Тогда
			глОшибкаID = 1;
			Сообщить("Двусмысленный ID: объект " + ТипСущ + " уже записан как " + ЧужойИд + ", новый " + Ид);
			Возврат;
		КонецЕсли;
	КонецЕсли;
	глТаблID.НоваяСтрока();
	глТаблID.entity_type = ТипСущ;
	глТаблID.id = Ид;
	глТаблID.вид = Вид;
	глТаблID.номер = Номер;
	глТаблID.дата = ДатаСтр;
	глТаблID.позиция = Позиция;
	глТаблID.номер_проводки = НомПров;
	глТаблID.ссылка = Ссылка;

Функция КУДиР_ПолеCSV(Знач С)
	С = "" + С;
	Кавычка = Симв(34);
	НужныКавычки = 0;
	Если Найти(С, ";") > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Кавычка) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Симв(10)) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Симв(13)) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если НужныКавычки = 1 Тогда
		С = СтрЗаменить(С, Кавычка, Кавычка + Кавычка);
		Возврат Кавычка + С + Кавычка;
	КонецЕсли;
	Возврат С;

Процедура КУДиР_СобратьОплаты(историяНач, датанач, датакон, таблОплаты, таблКлиенты)
	Опер = СоздатьОбъект("Операция");
	ДокОбъект = СоздатьОбъект("Документ");
	СписокПроводок = "50,62;51,62;55,62;";
	Опер.ВыбратьОперацииСПроводками(историяНач, датакон, СписокПроводок);
	Пока Опер.ПолучитьПроводку() = 1 Цикл
		Если КУДиР_ЭтоСчет62(Опер.Кредит.Счет) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Если КУДиР_ЭтоДенежныйСчет(Опер.Дебет.Счет) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Док = "";
		ДокСсылка = Опер.Документ;
		Если ДокОбъект.НайтиДокумент(ДокСсылка) = 1 Тогда
			Док = ДокОбъект.ТекущийДокумент();
		КонецЕсли;
		ВидДок = "";
		Если ПустоеЗначение(Док) = 0 Тогда
			ВидДок = СокрЛП(Док.Вид());
		КонецЕсли;
		Kind = КУДиР_ВидОплаты(ВидДок);
		Gran = "POSTING";
		НомПров = КУДиР_НомерПроводкиОпер(Опер);
		Если Kind = "CASH" Тогда
			Gran = "DOCUMENT";
			НомПровПоле = "";
		Иначе
			НомПровПоле = НомПров;
		КонецЕсли;
		payId = КУДиР_ИдОплаты(Kind, Док, Опер);
		Контр = Опер.Кредит.Контрагенты;
		Дог = Опер.Кредит.Договоры;
		ВидР = Опер.Кредит.ВидыРасчетовСПокупателями;
		cid = КУДиР_ИдКонтрагента(Контр);
		did = КУДиР_ИдДоговора(Контр, Дог);
		Поз = КУДиР_ПозицияДок(Док);
		ДтОпер = Опер.ДатаОперации;
		isT = "0";
		Если (ДтОпер >= датанач) И (ДтОпер <= датакон) Тогда
			isT = "1";
		КонецЕсли;
		КУДиР_ЗапомнитьID("PAYMENT", payId, Док, ВидДок, СокрЛП(Док.НомерДок), КУДиР_ДатаISO(Док.ДатаДок), Поз, НомПров);
		КУДиР_ЗапомнитьID("COUNTERPARTY", cid, Контр, "", СокрЛП(Контр.Код), "", "", "");
		КУДиР_ЗапомнитьID("CONTRACT", did, Дог, "", КУДиР_НомерДоговора(Дог), КУДиР_ДатаДоговора(Дог), "", "");
		Если isT = "1" Тогда
			КУДиР_ДобавитьКлиента(таблКлиенты, cid, Контр);
		КонецЕсли;
		коп = КУДиР_Копейки(Опер.Сумма);
		Если КУДиР_НайтиОплату(таблОплаты, payId) = 1 Тогда
			таблОплаты.СуммаКоп = Число(таблОплаты.СуммаКоп) + коп;
			Если таблОплаты.ДоговорID <> did Тогда
				таблОплаты.ДоговорID = "";
				таблОплаты.Договор = "";
				таблОплаты.НомерДоговора = "";
				таблОплаты.ДатаДоговора = "";
			КонецЕсли;
			Если таблОплаты.ВидРасчетовID <> КУДиР_ИдВидаРасчетов(ВидР) Тогда
				таблОплаты.ВидРасчетовID = "";
				таблОплаты.ВидРасчетовСПокупателем = "";
			КонецЕсли;
		Иначе
			таблОплаты.НоваяСтрока();
			таблОплаты.payment_id = payId;
			таблОплаты.is_target = isT;
			таблОплаты.PaymentKind = Kind;
			таблОплаты.PaymentGranularity = Gran;
			таблОплаты.ДатаОперации = КУДиР_ДатаISO(ДтОпер);
			таблОплаты.ПозицияДокумента = Поз;
			таблОплаты.НомерПроводкиВДокументе = НомПровПоле;
			таблОплаты.НомерСтрокиВДокументе = "";
			таблОплаты.ВидДокументаОплаты = ВидДок;
			таблОплаты.НомерДокументаОплаты = СокрЛП(Док.НомерДок);
			таблОплаты.ДатаДокументаОплаты = КУДиР_ДатаISO(Док.ДатаДок);
			таблОплаты.СуммаКоп = коп;
			таблОплаты.СчетДебета = СокрЛП("" + Опер.Дебет.Счет);
			таблОплаты.СчетКредита = СокрЛП("" + Опер.Кредит.Счет);
			таблОплаты.КонтрагентID = cid;
			таблОплаты.Контрагент = СокрЛП(Контр);
			таблОплаты.ДоговорID = did;
			таблОплаты.Договор = СокрЛП(Дог);
			таблОплаты.НомерДоговора = КУДиР_НомерДоговора(Дог);
			таблОплаты.ДатаДоговора = КУДиР_ДатаДоговора(Дог);
			таблОплаты.ВидРасчетовID = КУДиР_ИдВидаРасчетов(ВидР);
			таблОплаты.ВидРасчетовСПокупателем = СокрЛП(ВидР);
			таблОплаты.СодержаниеПроводки = СокрЛП(Опер.СодержаниеПроводки);
			таблОплаты.Содержание = СокрЛП(Опер.Содержание);
			таблОплаты.ПредставлениеПроводки = СокрЛП(Опер.ПредставлениеПроводки());
			таблОплаты.НазначениеПлатежа = СокрЛП(Опер.СодержаниеПроводки);
			таблОплаты.ПервичныйДокумент = СокрЛП(Опер.ПервичныйДокумент);
			таблОплаты.КомментарийДокументаОплаты = КУДиР_КомментарийДок(Док);
		КонецЕсли;
	КонецЦикла;
	н = таблОплаты.КоличествоСтрок();
	Пока н >= 1 Цикл
		cid = таблОплаты.ПолучитьЗначение(н, "КонтрагентID");
		Если КУДиР_КлиентВВыборке(таблКлиенты, cid) = 0 Тогда
			таблОплаты.УдалитьСтроку(н);
		КонецЕсли;
		н = н - 1;
	КонецЦикла;

Процедура КУДиР_ДобавитьСтрокуLedger(таблLedger, таблОплаты, Док, Опер, Знач Сторона, Знач Контр, Знач Дог, Знач ВидР, Знач Счет62)
	ВидДок = "";
	НомДок = "";
	ДтДок = "";
	Поз = "";
	Если ПустоеЗначение(Док) = 0 Тогда
		ВидДок = СокрЛП(Док.Вид());
		НомДок = СокрЛП(Док.НомерДок);
		ДтДок = КУДиР_ДатаISO(Док.ДатаДок);
		Поз = КУДиР_ПозицияДок(Док);
	КонецЕсли;
	НомПров = КУДиР_НомерПроводкиОпер(Опер);
	N = КУДиР_СледНомерКорр(Поз, НомПров);
	ledId = КУДиР_ИдLedger(Поз, НомПров, N);
	cid = КУДиР_ИдКонтрагента(Контр);
	did = КУДиР_ИдДоговора(Контр, Дог);
	payId = "";
	srcId = "";
	Kind = КУДиР_ВидОплаты(ВидДок);
	Если (Сторона = "K") И (КУДиР_ЭтоДенежныйСчет(Опер.Дебет.Счет) = 1) Тогда
		payId = КУДиР_ИдОплаты(Kind, Док, Опер);
		srcId = payId;

Процедура КУДиР_СобратьLedger(историяНач, датакон, таблОплаты, таблКлиенты, таблLedger)

	Опер.ВыбратьОперацииСПроводками(историяНач, датакон, СписокПроводок);

			Контр = Опер.Кредит.Контрагенты;
			cid = КУДиР_ИдКонтрагента(Контр);

			КонецЕсли;
		КонецЕсли;
	КонецЦикла;

Процедура КУДиР_ДобавитьДокумент(Док, таблДокументы, таблКлиенты)
	Если ПустоеЗначение(Док) = 1 Тогда
		Возврат;
	КонецЕсли;
	ВидДок = СокрЛП(Док.Вид());
	Role = КУДиР_РольДокумента(ВидДок);
	Если Role = "" Тогда
		Возврат;
	КонецЕсли;
	docId = КУДиР_ИдДокументаПозиция(Док);
	Если docId = "" Тогда
		Возврат;
	КонецЕсли;
	Если КУДиР_ЕстьИдВТаблице(таблДокументы, "document_id", docId) = 1 Тогда
		Возврат;
	КонецЕсли;
	Контр = КУДиР_КонтрагентДокумента(Док);
	cid = КУДиР_ИдКонтрагента(Контр);
	Если КУДиР_КлиентВВыборке(таблКлиенты, cid) = 0 Тогда
		Возврат;
	КонецЕсли;
	Дог = КУДиР_РеквизитДокумента(Док, "Договор");
	did = КУДиР_ИдДоговора(Контр, Дог);
	Поз = КУДиР_ПозицияДок(Док);
	КУДиР_ЗапомнитьID("DOCUMENT", docId, Док, ВидДок, СокрЛП(Док.НомерДок), КУДиР_ДатаISO(Док.ДатаДок), Поз, "");
	КУДиР_ЗапомнитьID("COUNTERPARTY", cid, Контр, "", СокрЛП(Контр.Код), "", "", "");
	Если ПустоеЗначение(Дог) = 0 Тогда
		КУДиР_ЗапомнитьID("CONTRACT", did, Дог, "", КУДиР_НомерДоговора(Дог), КУДиР_ДатаДоговора(Дог), "", "");
	КонецЕсли;
	таблДокументы.НоваяСтрока();
	таблДокументы.document_id = docId;
	таблДокументы.DocumentRole = Role;
	таблДокументы.DocumentType = КУДиР_ТипДокумента(ВидДок, Role);
	таблДокументы.ВидДокумента = ВидДок;
	таблДокументы.НомерДокумента = СокрЛП(Док.НомерДок);
	таблДокументы.ДатаДокумента = КУДиР_ДатаISO(Док.ДатаДок);
	таблДокументы.ПозицияДокумента = Поз;
	таблДокументы.КонтрагентID = cid;
	таблДокументы.Контрагент = СокрЛП(Контр);
	таблДокументы.ДоговорID = did;
	таблДокументы.Договор = СокрЛП(Дог);
	таблДокументы.НомерДоговора = КУДиР_НомерДоговора(Дог);
	таблДокументы.ДатаДоговора = КУДиР_ДатаДоговора(Дог);
	таблДокументы.ВидРасчетовID = "";
	таблДокументы.ВидРасчетовСПокупателем = "";
	таблДокументы.СуммаКоп = КУДиР_СуммаДокументаКоп(Док);
	таблДокументы.Проведен = КУДиР_ПроведенДок(Док);
	таблДокументы.ПомеченНаУдаление = КУДиР_ПомеченДок(Док);

	таблДокументы.Содержание = "";

Функция КУДиР_РазобратьCSVСтроку(Стр)
	сп = СоздатьОбъект("СписокЗначений");
	Поле = "";
	ВКавычках = 0;
	Кавычка = Симв(34);
	й = 1;
	Длн = СтрДлина(Стр);
	Пока й <= Длн Цикл
		Сим = Сред(Стр, й, 1);
		Если ВКавычках = 1 Тогда
			Если Сим = Кавычка Тогда
				Если й < Длн Тогда
					Если Сред(Стр, й + 1, 1) = Кавычка Тогда
						Поле = Поле + Кавычка;
						й = й + 1;
					Иначе
						ВКавычках = 0;
					КонецЕсли;
				Иначе
					ВКавычках = 0;
				КонецЕсли;
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		Иначе
			Если Сим = Кавычка Тогда
				ВКавычках = 1;
			ИначеЕсли Сим = ";" Тогда
				сп.ДобавитьЗначение(Поле);
				Поле = "";
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		КонецЕсли;
		й = й + 1;
	КонецЦикла;
	сп.ДобавитьЗначение(Поле);
	Возврат сп;

Функция КУДиР_ПрочитатьKV(Путь)
	таб = СоздатьОбъект("ТаблицаЗначений");
	таб.НоваяКолонка("key");
	таб.НоваяКолонка("value");
	FSO = СоздатьОбъект("Scripting.FileSystemObject");
	Если FSO.FileExists(Путь) = 0 Тогда
		Возврат таб;
	КонецЕсли;
	TS = FSO.OpenTextFile(Путь, 1, 0);
	Первая = 1;
	Пока TS.AtEndOfStream = 0 Цикл
		Стр = TS.ReadLine();
		Если Первая = 1 Тогда
			Первая = 0;
			Продолжить;
		КонецЕсли;
		сп = КУДиР_РазобратьCSVСтроку(Стр);
		таб.НоваяСтрока();
		таб.key = КУДиР_ПолеСписка(сп, 1);
		таб.value = КУДиР_ПолеСписка(сп, 2);
	КонецЦикла;
	TS.Close();
	Возврат таб;

		Если таб.key = Ключ Тогда
			Возврат таб.value;

	КонецЦикла;
	Возврат "";

Функция КУДиР_ЗагрузитьИВосстановить()
	таб = СформироватьТаблицуДоходовЧистовая();
	маниф = КУДиР_ПрочитатьKV(глКаталогОбмена + "\manifest.csv");

	Статус = СокрЛП(КУДиР_ЗначениеKV(стат, "status"));
	runСтат = СокрЛП(КУДиР_ЗначениеKV(стат, "run_id"));
	taxReady = СокрЛП(КУДиР_ЗначениеKV(стат, "tax_ready"));
	unresCnt = СокрЛП(КУДиР_ЗначениеKV(стат, "unresolved_count"));
	unresKop = СокрЛП(КУДиР_ЗначениеKV(стат, "unresolved_debt_kopecks"));
	Ошибка = КУДиР_ЗначениеKV(стат, "error");
	Если Статус = "" Тогда
		Статус = "FAILED";
		Если Ошибка = "" Тогда
			Ошибка = "нет run_status.csv - Python не запустился. Смотрите python_run.log";
		КонецЕсли;
	КонецЕсли;
	Сообщить("КУДиР: status=" + Статус + " tax_ready=" + taxReady);
	Если Статус = "FAILED" Тогда
		Сообщить("Ошибка: " + Ошибка);
		Возврат таб;
	КонецЕсли;
	Если (Статус <> "SUCCESS") И (Статус <> "SUCCESS_DEGRADED") Тогда
		Сообщить("КУДиР: неизвестный status=" + Статус + ", результат не загружается.");
		Возврат таб;
	КонецЕсли;
	Если Статус = "SUCCESS_DEGRADED" Тогда
		Сообщить("Предупреждение: matching в ухудшенном режиме (parser недоступен). Строки HIGH загружаются.");
	КонецЕсли;
	Если runМаниф = "" Тогда
		Сообщить("КУДиР: в manifest.csv нет run_id, результат не загружается.");
		Возврат таб;
	КонецЕсли;
	Если runСтат <> runМаниф Тогда
		Сообщить("КУДиР: run_id run_status (" + runСтат + ") не совпал с manifest (" + runМаниф + "). Чужой результат не загружается.");
		Возврат таб;
	КонецЕсли;
	Если taxReady <> "1" Тогда
		Сообщить("КУДиР не готова: tax_ready=" + taxReady + ", unresolved_count=" + unresCnt + ", unresolved_debt_kopecks=" + unresKop + ". Matching загружается, цикл 2 не стартовать.");
	КонецЕсли;

	Путь = глКаталогОбмена + "\kudir_result.csv";
	FSO = СоздатьОбъект("Scripting.FileSystemObject");
	Если FSO.FileExists(Путь) = 0 Тогда
		Сообщить("Нет kudir_result.csv");
		Возврат таб;
	КонецЕсли;
	TS = FSO.OpenTextFile(Путь, 1, 0);
	Если TS.AtEndOfStream = 1 Тогда
		TS.Close();
		Возврат таб;
	КонецЕсли;
	спЗаг = КУДиР_РазобратьCSVСтроку(TS.ReadLine());
	нRun = КУДиР_НомерКолонки(спЗаг, "run_id");
	нPay = КУДиР_НомерКолонки(спЗаг, "payment_id");
	нDoc = КУДиР_НомерКолонки(спЗаг, "matched_document_id");
	нDog = КУДиР_НомерКолонки(спЗаг, "contract_id");
	нSum = КУДиР_НомерКолонки(спЗаг, "amount_kopecks");
	нAdv = КУДиР_НомерКолонки(спЗаг, "advance_kopecks");
	нRep = КУДиР_НомерКолонки(спЗаг, "repayment_kopecks");
	нInc = КУДиР_НомерКолонки(спЗаг, "income_kopecks");
	нVatB = КУДиР_НомерКолонки(спЗаг, "vat_base_kopecks");
	нVat = КУДиР_НомерКолонки(спЗаг, "vat_kopecks");

		КонецЕсли;
