		Возврат "";
	КонецЕсли;
	Г = "" + ДатаГод(Д);

		М = "0" + М;
	КонецЕсли;
	Чи = "" + ДатаЧисло(Д);
	Если СтрДлина(Чи) = 1 Тогда
		Чи = "0" + Чи;
	КонецЕсли;
	Возврат Г + "-" + М + "-" + Чи;

Функция КУДиР_ПолеCSV(Знач С)
	С = "" + С;
	Кавычка = Симв(34);
	НужныКавычки = 0;
	Если Найти(С, ";") > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Кавычка) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Симв(10)) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если Найти(С, Симв(13)) > 0 Тогда
		НужныКавычки = 1;
	КонецЕсли;
	Если НужныКавычки = 1 Тогда
		С = СтрЗаменить(С, Кавычка, Кавычка + Кавычка);
		Возврат Кавычка + С + Кавычка;
	КонецЕсли;
	Возврат С;

Процедура КУДиР_СобратьОплаты(датанач, датакон, таблОплаты, таблLedger)
	Опер = СоздатьОбъект("Операция");
	СписокПроводок = "50,62;51,62;55,62;";
	Опер.ВыбратьОперацииСПроводками(датанач, датакон, СписокПроводок);
	Пока Опер.ПолучитьПроводку() = 1 Цикл
		Если КУДиР_ЭтоСчет62(Опер.Кредит.Счет) = 0 Тогда
			Продолжить;
		КонецЕсли;
		Док = Опер.Документ;
		ВидДок = "";
		Если ПустоеЗначение(Док) = 0 Тогда
			ВидДок = СокрЛП(Док.Вид());
		КонецЕсли;
		Контр = Опер.Кредит.Контрагенты;
		Дог = Опер.Кредит.Договоры;
		ВидР = Опер.Кредит.ВидыРасчетовСПокупателями;
		НомСтр = КУДиР_НомерСтрокиОпер(Опер, ВидДок);
		ИдДок = КУДиР_ИдДокумента(Док);
		Kind = "BANK";

Функция КУДиР_РазобратьCSVСтроку(Стр)
	сп = СоздатьОбъект("СписокЗначений");
	Поле = "";
	ВКавычках = 0;
	Кавычка = Симв(34);
	й = 1;
	Длн = СтрДлина(Стр);
	Пока й <= Длн Цикл
		Сим = Сред(Стр, й, 1);
		Если ВКавычках = 1 Тогда
			Если Сим = Кавычка Тогда
				Если й < Длн Тогда
					Если Сред(Стр, й + 1, 1) = Кавычка Тогда
						Поле = Поле + Кавычка;
						й = й + 1;
					Иначе
						ВКавычках = 0;
					КонецЕсли;
				Иначе
					ВКавычках = 0;
				КонецЕсли;
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		Иначе
			Если Сим = Кавычка Тогда
				ВКавычках = 1;
			ИначеЕсли Сим = ";" Тогда
				сп.ДобавитьЗначение(Поле);
				Поле = "";
			Иначе
				Поле = Поле + Сим;
			КонецЕсли;
		КонецЕсли;
		й = й + 1;
	КонецЦикла;
	сп.ДобавитьЗначение(Поле);
	Возврат сп;

Функция КУДиР_ПолеСписка(сп, Ном)
	Если Ном < 1 Тогда
		Возврат "";
	КонецЕсли;
	Если Ном > сп.Количество() Тогда
		Возврат "";
	КонецЕсли;
	Возврат "" + сп.ПолучитьЗначение(Ном);

Функция КУДиР_НомерКолонки(спЗаг, Имя)
	й = 1;
	Пока й <= спЗаг.Количество() Цикл
		Если спЗаг.ПолучитьЗначение(й) = Имя Тогда
			Возврат й;
		КонецЕсли;
		й = й + 1;
	КонецЦикла;
	Возврат 0;

Функция КУДиР_ПрочитатьKV(Путь)
	таб = СоздатьОбъект("ТаблицаЗначений");
	таб.НоваяКолонка("key");
	таб.НоваяКолонка("value");
	FSO = СоздатьОбъект("Scripting.FileSystemObject");
	Если FSO.FileExists(Путь) = 0 Тогда
		Возврат таб;
	КонецЕсли;
	TS = FSO.OpenTextFile(Путь, 1, 0);
	Первая = 1;
	Пока TS.AtEndOfStream = 0 Цикл
		Стр = TS.ReadLine();
		Если Первая = 1 Тогда
			Первая = 0;
			Продолжить;
		КонецЕсли;
		сп = КУДиР_РазобратьCSVСтроку(Стр);
		таб.НоваяСтрока();
		таб.key = КУДиР_ПолеСписка(сп, 1);
		таб.value = КУДиР_ПолеСписка(сп, 2);
	КонецЦикла;
	TS.Close();
	Возврат таб;
КонецФункции

Функция КУДиР_ЗначениеKV(таб, Ключ)
	таб.ВыбратьСтроки();
	Пока таб.ПолучитьСтроку() = 1 Цикл
		Если таб.key = Ключ Тогда
			Возврат таб.value;
		КонецЕсли;
	КонецЦикла;
	Возврат "";

Функция КУДиР_ЗагрузитьИВосстановить()
	таб = СформироватьТаблицуДоходовЧистовая();
	стат = КУДиР_ПрочитатьKV(глКаталогОбмена + "\run_status.csv");
	Статус = КУДиР_ЗначениеKV(стат, "status");
	Если Статус = "" Тогда
		Статус = "FAILED";
	КонецЕсли;
	Сообщить("Прототип КУДиР: status=" + Статус);
	Если Статус = "FAILED" Тогда
		Сообщить("Ошибка: " + КУДиР_ЗначениеKV(стат, "error"));
		Возврат таб;
	КонецЕсли;
	Если Статус = "SUCCESS_DEGRADED" Тогда
		Сообщить("Предупреждение: анализ в ухудшенном режиме (parser недоступен).");
	КонецЕсли;

	Путь = глКаталогОбмена + "\kudir_result.csv";
	FSO = СоздатьОбъект("Scripting.FileSystemObject");
	Если FSO.FileExists(Путь) = 0 Тогда
		Сообщить("Нет kudir_result.csv");
		Возврат таб;
	КонецЕсли;
	TS = FSO.OpenTextFile(Путь, 1, 0);
	Если TS.AtEndOfStream = 1 Тогда
		TS.Close();
		Возврат таб;
	КонецЕсли;
	спЗаг = КУДиР_РазобратьCSVСтроку(TS.ReadLine());
	нPay = КУДиР_НомерКолонки(спЗаг, "payment_id");
	нDoc = КУДиР_НомерКолонки(спЗаг, "matched_document_id");
	нDog = КУДиР_НомерКолонки(спЗаг, "contract_id");
	нSum = КУДиР_НомерКолонки(спЗаг, "amount_kopecks");
	нAdv = КУДиР_НомерКолонки(спЗаг, "advance_kopecks");
	нRep = КУДиР_НомерКолонки(спЗаг, "repayment_kopecks");
	нInc = КУДиР_НомерКолонки(спЗаг, "income_kopecks");
	нVatB = КУДиР_НомерКолонки(спЗаг, "vat_base_kopecks");
	нVat = КУДиР_НомерКолонки(спЗаг, "vat_kopecks");
	нTxt = КУДиР_НомерКолонки(спЗаг, "СодержаниеЗаписи");
	нType = КУДиР_НомерКолонки(спЗаг, "row_type");

	Ошибок = 0;
	Ок = 0;
	Пока TS.AtEndOfStream = 0 Цикл
		сп = КУДиР_РазобратьCSVСтроку(TS.ReadLine());
		payId = КУДиР_ПолеСписка(сп, нPay);
		docId = КУДиР_ПолеСписка(сп, нDoc);
		dogId = КУДиР_ПолеСписка(сп, нDog);
		ДокОпл = КУДиР_НайтиСсылку("PAYMENT", payId);
		Договор = КУДиР_НайтиСсылку("CONTRACT", dogId);
		ДокРеал = КУДиР_НайтиСсылку("DOCUMENT", docId);

		Если ПустоеЗначение(ДокОпл) = 1 Тогда
			Сообщить("НЕ восстановлен ДокументОплаты payment_id=" + payId);
			Ошибок = Ошибок + 1;
		Иначе
			Сообщить("OK ДокументОплаты " + payId + " -> " + СокрЛП(ДокОпл));
			Ок = Ок + 1;
		КонецЕсли;
		Если СокрЛП(dogId) <> "" Тогда
			Если ПустоеЗначение(Договор) = 1 Тогда
				Сообщить("НЕ восстановлен Договор contract_id=" + dogId);
				Ошибок = Ошибок + 1;
			Иначе
				Сообщить("OK Договор " + dogId + " -> " + СокрЛП(Договор));
			КонецЕсли;
		КонецЕсли;
		Если СокрЛП(docId) <> "" Тогда
			Если ПустоеЗначение(ДокРеал) = 1 Тогда
				Сообщить("НЕ восстановлен документ реализации matched_document_id=" + docId);
				Ошибок = Ошибок + 1;
			Иначе
				Сообщить("OK Документ реализации " + docId + " -> " + СокрЛП(ДокРеал));
			КонецЕсли;
		КонецЕсли;

		таб.НоваяСтрока();
		таб.ДокументОплаты = ДокОпл;
		Попытка
			таб.НомерДокументаОплаты = СокрЛП(ДокОпл.Номер);
		Исключение
			таб.НомерДокументаОплаты = "";
		КонецПопытки;
		Попытка
			таб.ДатаПлатежа = ДокОпл.ДатаДок;
		Исключение
			таб.ДатаПлатежа = "";
		КонецПопытки;
		таб.СуммаПлатежа = Число(КУДиР_ПолеСписка(сп, нSum)) / 100;
		таб.Договор = Договор;
		Попытка
			таб.НомерДоговора = СокрЛП(Договор.Номер);
		Исключение
			таб.НомерДоговора = СокрЛП(Договор);
		КонецПопытки;
		таб.СодержаниеЗаписи = КУДиР_ПолеСписка(сп, нTxt);
		таб.СуммаВАванс = Число(КУДиР_ПолеСписка(сп, нAdv)) / 100;
		таб.СуммаВОпалату = Число(КУДиР_ПолеСписка(сп, нRep)) / 100;

		таб.СуммаВДоход = Число(КУДиР_ПолеСписка(сп, нInc)) / 100;
	КонецЦикла;
	TS.Close();
	Сообщить("Прототип КУДиР: восстановлено ссылок оплат OK=" + СокрЛП(Ок) + ", ошибок=" + СокрЛП(Ошибок));
	Возврат таб;
